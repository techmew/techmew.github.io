/************************************
 * ひまなねこのひまつぶし用
 * Slack → GAS → はてなブログ半自動投稿
 *
 * 完成仕様：
 * 1. Slackに1〜10件のarticle.txtだけを含む文章ZIPを送る
 * 2. GASが各IMAGE_PROMPTを検査し、復旧・h3照合用としてDrive内部へ個別txtとmanifest.txtを保存する
 * 3. GASは全IMAGE_PROMPTをSLOT_01〜Nへ展開し、HNK_EMBEDDED_INDEPENDENT_V2の自己完結した画像生成依頼をSlackへ返す。画像生成時のDrive再読は不要
 * 4. 承認済みPNGが揃った後、h3自動がmanifestのIMAGE_FILE_NAMEへ対応付けて画像ZIPを作成する
 * 5. 指定ファイル名のPNGだけを含む画像ZIPをSlackへ送る
 * 6. GASがBATCH_ID / WORK_ID / IMAGE_FILE_NAMEで文章と画像を照合する
 * 7. 画像を記事トップに挿入して全件公開する
 * 8. Slackには公開済み記事URL一覧を返す
 *
 * 注意：
 * ・LINEは使いません
 * ・GASからXへ自動投稿はしません
 * ・X_POSTは固定タグとして読み取りますが、Slack返信には使用しません
 * ・旧タグ META / X_POST_GAS / CHECK は使いません
 * ・こまちのゲームナビ / みはりちゃんのガジェメモ等、別ブログ情報が混ざった場合は処理せずSlackに警告します
 * ・入力タグは TITLE / CATEGORY / META_DESCRIPTION / HTML / X_POST / IMAGE_PROMPT が必須です
 * ・Slack画像のContent-Typeが application/octet-stream になる問題を補正します
 ************************************/

/**
 * Webhook入口
 */
function driveDoPost(e) {
  try {
    if (!e || !e.postData || !e.postData.contents) {
      return ContentService.createTextOutput('NO_POST_DATA');
    }
















    const body = JSON.parse(e.postData.contents);
















    // Slack Request URL認証
    if (body.type === 'url_verification' && body.challenge) {
      return ContentService
        .createTextOutput(body.challenge)
        .setMimeType(ContentService.MimeType.TEXT);
    }
















    // Slack Events API
    if (body.type === 'event_callback' && body.event) {
      const eventId = body.event_id || body.event.event_ts || body.event.ts || '';
















      // SlackはWebhook応答が遅いと同じevent_idを再送する。
      // 処理開始前にLockService内で原子的にclaimし、同一イベントの並行実行を止める。
      if (eventId && !claimSlackEvent_(eventId)) {
        return ContentService.createTextOutput('OK');
      }
















      try {
        handleSlackEvent(body.event);
        if (eventId) completeSlackEvent_(eventId, 'done', '');
      } catch (eventErr) {
        if (eventId) {
          completeSlackEvent_(
            eventId,
            'failed',
            String(eventErr && eventErr.message ? eventErr.message : eventErr)
          );
        }
        throw eventErr;
      }
      return ContentService.createTextOutput('OK');
    }
















    return ContentService.createTextOutput('OK');
  } catch (err) {
    console.error(err);
    return ContentService.createTextOutput('ERROR: ' + err.message);
  }
}
















/**
 * Slackイベントを原子的にclaimする。
 * done済み、または有効期限内のprocessingは重複として拒否する。
 */
function claimSlackEvent_(eventId) {
  if (!eventId) return true;
  const lock = LockService.getScriptLock();
  lock.waitLock(5000);
  try {
    const props = PropertiesService.getScriptProperties();
    const key = SLACK_EVENT_STATE_PREFIX + eventId;
    const now = Date.now();
    const current = parseJsonSafe_(props.getProperty(key), {});
    const age = now - Number(current.ts || 0);
















    if (current.status === 'done' && age < SLACK_EVENT_DONE_RETENTION_MS) return false;
    if (current.status === 'processing' && age < SLACK_EVENT_PROCESSING_TTL_MS) return false;
















    setScriptPropertySafe_(key, JSON.stringify({ status: 'processing', ts: now }));
    return true;
  } finally {
    lock.releaseLock();
  }
}
































/**
 * Slackイベントの処理結果を保存する。
 */
function completeSlackEvent_(eventId, status, errorMessage) {
  if (!eventId) return;
  const lock = LockService.getScriptLock();
  lock.waitLock(5000);
  try {
    setScriptPropertySafe_(
      SLACK_EVENT_STATE_PREFIX + eventId,
      JSON.stringify({
        status: status || 'done',
        ts: Date.now(),
        error: String(errorMessage || '')
      })
    );
  } finally {
    lock.releaseLock();
  }
  cleanupOldSlackEventKeys_();
}

  if (!eventId) return false;
  const props = PropertiesService.getScriptProperties();
  if (props.getProperty('SLACK_EVENT_DONE_' + eventId)) return true;
















  const current = parseJsonSafe_(props.getProperty(SLACK_EVENT_STATE_PREFIX + eventId), {});
  const age = Date.now() - Number(current.ts || 0);
  return (
    (current.status === 'done' && age < SLACK_EVENT_DONE_RETENTION_MS) ||
    (current.status === 'processing' && age < SLACK_EVENT_PROCESSING_TTL_MS)
  );
}

function parseJsonSafe_(raw, fallback) {
  if (!raw) return fallback;
  try {
    return JSON.parse(raw);
  } catch (err) {
    return fallback;
  }
}








/**
 * Script Propertiesの容量超過を自己復旧する。
 * 認証情報・キュー本体・PENDING_BATCHは削除せず、重複防止用の一時キーだけを整理する。
 */
function isScriptPropertyQuotaError_(err) {
  const message = String(err && err.message ? err.message : err || '');
  return /property storage quota|exceeded the property storage quota/i.test(message);
}




function getScriptPropertyTimestamp_(key, value) {
  if (!value) return 0;




  if (String(key || '').indexOf('SLACK_EVENT_DONE_') === 0) {
    return Number(value || 0) || 0;
  }




  const parsed = parseJsonSafe_(value, null);
  if (parsed && typeof parsed === 'object') {
    const rawTs = parsed.ts || parsed.updatedAt || parsed.createdAt || '';
    if (typeof rawTs === 'number') return rawTs;
    const numeric = Number(rawTs || 0);
    if (numeric) return numeric;
    const dateTs = Date.parse(String(rawTs || ''));
    if (!isNaN(dateTs)) return dateTs;
  }




  const directNumeric = Number(value || 0);
  if (directNumeric) return directNumeric;
  const directDateTs = Date.parse(String(value || ''));
  return isNaN(directDateTs) ? 0 : directDateTs;
}




function isTransientScriptPropertyKey_(key) {
  key = String(key || '');
  return key.indexOf('SLACK_EVENT_DONE_') === 0 ||
    key.indexOf(SLACK_EVENT_STATE_PREFIX) === 0 ||
    key.indexOf(SLACK_FILE_STATE_PREFIX) === 0 ||
    key.indexOf(IMAGE_BATCH_ACCEPT_PREFIX) === 0 ||
    key.indexOf(TEXT_BATCH_ACCEPT_PREFIX) === 0 ||
    key.indexOf(ZIP_BATCH_HASH_PREFIX) === 0 ||
    key.indexOf(ZIP_ARTICLE_HASH_PREFIX) === 0 ||
    key.indexOf(IMAGE_ZIP_HASH_PREFIX) === 0;
}




function isActiveTransientScriptProperty_(key, value, now) {
  const state = parseJsonSafe_(value, {});
  if (String(state.status || '') !== 'processing') return false;
  const ts = getScriptPropertyTimestamp_(key, value);
  const age = ts ? now - ts : Number.MAX_SAFE_INTEGER;




  if (String(key || '').indexOf(SLACK_EVENT_STATE_PREFIX) === 0) {
    return age < SLACK_EVENT_PROCESSING_TTL_MS;
  }
  if (String(key || '').indexOf(SLACK_FILE_STATE_PREFIX) === 0) {
    return age < SLACK_FILE_PROCESSING_TTL_MS;
  }
  if (String(key || '').indexOf(IMAGE_BATCH_ACCEPT_PREFIX) === 0) {
    return age < IMAGE_BATCH_PROCESSING_TTL_MS;
  }
  if (String(key || '').indexOf(TEXT_BATCH_ACCEPT_PREFIX) === 0) {
    return age < SLACK_FILE_PROCESSING_TTL_MS;
  }
  return false;
}




function cleanupScriptPropertyQuota_(aggressive) {
  const props = PropertiesService.getScriptProperties();
  const before = props.getProperties();
  const now = Date.now();




  // 容量超過時は一時キーを個別削除せず、保持対象だけを一括で残す。
  // setProperties(..., true) で置換するため、大量deletePropertyによる実行時間超過を避ける。
  if (aggressive) {
    const keep = {};
    Object.keys(before).forEach(function(key) {
      if (isTransientScriptPropertyKey_(key)) return;
      if (String(key || '').indexOf('ONE_SHOT_') === 0) return;
      if (key === SLACK_TEXT_RECOVERY_LAST_RUN_KEY ||
          key === SLACK_IMAGE_RECOVERY_LAST_RUN_KEY ||
          key === SLACK_IMAGE_RECOVERY_CURSOR_KEY ||
          key === 'AUTH_TEST') return;
      keep[key] = before[key];
    });




    props.setProperties(keep, true);
    const afterBulk = props.getProperties();
    return {
      beforeKeys: Object.keys(before).length,
      afterKeys: Object.keys(afterBulk).length,
      transientBefore: Object.keys(before).filter(isTransientScriptPropertyKey_).length,
      deleted: Object.keys(before).length - Object.keys(afterBulk).length
    };
  }




  // 通常時は期限超過だけを掃除し、必要ならキー数上限まで古い終了済み一時キーを削る。
  cleanupOldSlackEventKeys_();
  cleanupOldZipBatchHashKeys_();




  const all = props.getProperties();
  const transient = Object.keys(all)
    .filter(isTransientScriptPropertyKey_)
    .map(function(key) {
      return {
        key: key,
        value: all[key],
        ts: getScriptPropertyTimestamp_(key, all[key]),
        active: isActiveTransientScriptProperty_(key, all[key], now)
      };
    });




  const removable = transient
    .filter(function(entry) { return !entry.active; })
    .sort(function(a, b) {
      if (!a.ts && !b.ts) return a.key.localeCompare(b.key);
      if (!a.ts) return -1;
      if (!b.ts) return 1;
      return a.ts - b.ts;
    });




  let transientCount = transient.length;
  let deleted = 0;
  for (let i = 0; i < removable.length && transientCount > SCRIPT_PROPERTY_TRANSIENT_MAX_KEYS; i++) {
    props.deleteProperty(removable[i].key);
    transientCount--;
    deleted++;
  }




  const after = props.getProperties();
  return {
    beforeKeys: Object.keys(before).length,
    afterKeys: Object.keys(after).length,
    transientBefore: transient.length,
    deleted: deleted
  };
}




function setScriptPropertySafe_(key, value) {
  const props = PropertiesService.getScriptProperties();
  try {
    props.setProperty(String(key), String(value));
    return;
  } catch (err) {
    if (!isScriptPropertyQuotaError_(err)) throw err;
  }




  cleanupScriptPropertyQuota_(true);
  props.setProperty(String(key), String(value));
}








/**
 * 古いSlackイベント／ファイルclaimを掃除する。
 */
function cleanupOldSlackEventKeys_() {
  const props = PropertiesService.getScriptProperties();
  const all = props.getProperties();
  const now = Date.now();
















  Object.keys(all).forEach(key => {
    const value = all[key];
















    if (key.startsWith('SLACK_EVENT_DONE_')) {
      const t = Number(value || 0);
      if (t && now - t > SLACK_EVENT_DONE_RETENTION_MS) props.deleteProperty(key);
      return;
    }
















    if (key.startsWith(SLACK_EVENT_STATE_PREFIX)) {
      const state = parseJsonSafe_(value, {});
      const age = now - Number(state.ts || 0);
      if (age > SLACK_EVENT_DONE_RETENTION_MS) props.deleteProperty(key);
      return;
    }
















    if (key.startsWith(SLACK_FILE_STATE_PREFIX)) {
      const state = parseJsonSafe_(value, {});
      const age = now - Number(state.ts || 0);
      if (age > SLACK_FILE_DONE_RETENTION_MS) props.deleteProperty(key);
      return;
    }
















    if (key.startsWith(IMAGE_BATCH_ACCEPT_PREFIX) || key.startsWith(TEXT_BATCH_ACCEPT_PREFIX)) {
      const state = parseJsonSafe_(value, {});
      const age = now - Number(state.ts || 0);
      if (age > SLACK_FILE_DONE_RETENTION_MS) props.deleteProperty(key);
    }
  });
}
































/**
 * Slack Event処理
 */
function handleSlackEvent(event) {
  try {
    if (!event) return;
    if (event.type !== 'message') return;
    const rawText = String(event.text || '').trim();
    // ChatGPT経由のSlack投稿では、本文の前後にメンション・装飾・送信元表示が混ざることがある。
    // 先頭行からSlackメンションと装飾記号を除去し、既知コマンドが行頭にある場合だけコマンド扱いする。
    const firstLine = String(rawText.split(/\r?\n/, 1)[0] || '').trim();
    const cleanedCommandLine = firstLine
      .replace(/<@[A-Z0-9]+(?:\|[^>]+)?>/gi, ' ')
      .replace(/[*_`~]/g, ' ')
      .replace(/\s+/g, ' ')
      .trim();
    const knownCommands = [
      '記事台帳同期', '台帳同期', '記事台帳確認', '台帳確認',
      'はてな疎通テスト', '疎通テスト', '画像プロンプトURL', 'Driveプロンプト', 'プロンプトURL',
      '文章ZIP再処理', '文章再処理', 'ZIP再処理', '文章Drive再処理', '画像Drive再処理', '画像再処理', '再処理',
      'プロパティ掃除', '設定確認', 'バージョン', 'version', 'バッチ再開', 'H0G再開', 'リセット', 'reset', '公開:', '公開', '状態', 'テスト', 'ガス'
    ];
    const matchedCommand = knownCommands.find(function(command) {
      return cleanedCommandLine === command || cleanedCommandLine.indexOf(command + ' ') === 0;
    });
    // 引数付きDrive復旧コマンドはknownCommands正規化で引数を落とさない。
    const rawDriveTextRetryMatch = firstLine.match(
      /^文章Drive再処理[：:\s]+([A-Za-z0-9_-]+)/
    );
    const rawDriveImageRetryMatch = firstLine.match(
      /^画像Drive再処理[：:\s]+([A-Za-z0-9_-]+)/
    );




    // ChatGPT経由投稿では「使用して送信されました」等の帰属表示が
    // コマンドと同じイベント本文へ連結される場合がある。
    // バッチ再開だけは行全体の完全一致に依存せず、先頭のBATCH_IDを正本として抽出する。
    // BATCH_IDにはアンダースコアを含むため、Markdown装飾除去より前の先頭行を最優先で解析する。
    // cleanedCommandLineでは '_' を装飾記号として除去するため、BATCH_IDがH0Gだけに切れる事故を防ぐ。
    const rawBatchResumeMatch = firstLine.match(
      /^バッチ再開[：:\s]+([A-Za-z0-9_-]+)(?:\s+([A-Za-z0-9_-]+))?/
    );
    const rawBatchStatusMatch = firstLine.match(
      /^状態[：:\s]+([A-Za-z0-9_-]+)/
    );
    const exactBatchResumeMatch = cleanedCommandLine.match(
      /^バッチ再開[：:\s]+([A-Za-z0-9_-]+)(?:\s+([A-Za-z0-9_-]+))?$/
    );
    const prefixBatchResumeMatch = cleanedCommandLine.match(
      /^バッチ再開[：:\s]+([A-Za-z0-9_-]+)/
    );
    const normalizedBatchResumeMatch = rawBatchResumeMatch || exactBatchResumeMatch || prefixBatchResumeMatch;
    const commandText = normalizedBatchResumeMatch
      ? ('バッチ再開 ' + normalizedBatchResumeMatch[1] +
         (exactBatchResumeMatch && exactBatchResumeMatch[2] ? ' ' + exactBatchResumeMatch[2] : ''))
      : (rawBatchStatusMatch
          ? ('状態 ' + rawBatchStatusMatch[1])
          : (rawDriveImageRetryMatch
          ? ('画像Drive再処理 ' + rawDriveImageRetryMatch[1])
          : (rawDriveTextRetryMatch
              ? ('文章Drive再処理 ' + rawDriveTextRetryMatch[1])
              : (matchedCommand || cleanedCommandLine))));
    const allowBotLedgerCommand = /^(記事台帳同期|台帳同期|記事台帳確認|台帳確認)$/.test(commandText)
      || Boolean(normalizedBatchResumeMatch)
      || Boolean(rawBatchStatusMatch)
      || Boolean(rawDriveImageRetryMatch)
      || Boolean(rawDriveTextRetryMatch);
    if (event.bot_id && !allowBotLedgerCommand) return;
































    const channel = event.channel;
































    const allowedChannel = getOptionalProp(CONFIG_KEYS.SLACK_CHANNEL_ID);
    if (allowedChannel && channel !== allowedChannel) {
      return;
    }




    // パラメータ付きのバッチ再開は、汎用コマンド正規化より前に直接処理する。
    // 帰属表示などの末尾ノイズが付いても、先頭のBATCH_IDで再開する。
    const directBatchResumeMatch = normalizedBatchResumeMatch;
    if (directBatchResumeMatch) {
      const resumeThreadTs = event.thread_ts || event.ts || '';
      try {
        const resumed = resumeZipBatchById_(
          directBatchResumeMatch[1],
          channel,
          resumeThreadTs,
          exactBatchResumeMatch && exactBatchResumeMatch[2] ? exactBatchResumeMatch[2] : ''
        );
        postSlackMessage(
          channel,
          'バッチ再開受付\nBATCH_ID：' + resumed.batchId +
          '\n未処理：' + resumed.pendingCount + '件' +
          '\n失敗から復旧：' + resumed.resetCount + '件' +
          '\n公開済み：' + resumed.publishedCount + '件' +
          '\n完了通知：元TEXT ZIPスレッドへ集約',
          resumeThreadTs
        );
      } catch (err) {
        postSlackMessage(
          channel,
          'バッチ再開エラー：' + String(err && err.message ? err.message : err),
          resumeThreadTs
        );
      }
      return;
    }
































    const text = rawText;
































    if (event.files && event.files.length > 0) {
      handleSlackFileMessage(channel, event.files, event.thread_ts || event.ts || '');
      return;
    }
































    if (!text) return;
































    if (commandText === 'ガス') {
      postSlackMessage(channel, 'これはドライブから読み込んでるよ');
      return;
    }
































    if (commandText === 'テスト') {
      postSlackMessage(channel, 'OK');
      return;
    }
































    if (commandText === 'プロパティ掃除') {
      try {
        const cleanedProps = cleanupScriptPropertyQuota_(true);
        postSlackMessage(
          channel,
          'Script Properties掃除完了\n全キー：' + cleanedProps.beforeKeys + ' → ' + cleanedProps.afterKeys +
          '\n一時キー：' + cleanedProps.transientBefore + '件\n削除：' + cleanedProps.deleted + '件',
          event.thread_ts || event.ts || ''
        );
      } catch (cleanupErr) {
        postSlackMessage(
          channel,
          '❌ Script Properties掃除エラー：' + String(cleanupErr && cleanupErr.message ? cleanupErr.message : cleanupErr),
          event.thread_ts || event.ts || ''
        );
      }
      return;
    }








    if (commandText === '設定確認') {
      postSlackMessage(channel, getPropertiesStatusText());
      return;
    }
































    if (commandText === 'バージョン' || commandText.toLowerCase() === 'version') {
      postSlackMessage(channel, 'GAS_VERSION: ' + SCRIPT_VERSION);
      return;
    }








    if (commandText === 'H0G再開') {
      const resumeThreadTs = event.thread_ts || event.ts || '';
      try {
        const resumed = resumeZipBatchById_(
          'H0G_20260824_01',
          channel,
          resumeThreadTs,
          '1fTodVUEbWzDNeDqxxMurl6W67222DHWU'
        );
        postSlackMessage(
          channel,
          'バッチ再開受付\nBATCH_ID：' + resumed.batchId +
          '\n未処理：' + resumed.pendingCount + '件' +
          '\n失敗から復旧：' + resumed.resetCount + '件' +
          '\n公開済み：' + resumed.publishedCount + '件' +
          '\n完了通知：元TEXT ZIPスレッドへ集約',
          resumeThreadTs
        );
      } catch (err) {
        postSlackMessage(
          channel,
          'H0G再開エラー：' + String(err && err.message ? err.message : err),
          resumeThreadTs
        );
      }
      return;
    }
































    if (commandText === '記事台帳同期' || commandText === '台帳同期') {
      handleArticleLedgerSyncCommand_(channel, event.thread_ts || event.ts || '');
      return;
    }
































    if (commandText === '記事台帳確認' || commandText === '台帳確認') {
      handleArticleLedgerStatusCommand_(channel, event.thread_ts || event.ts || '');
      return;
    }
































    if (commandText === 'はてな疎通テスト' || commandText === '疎通テスト') {
      handleHatenaConnectionTest(channel);
      return;
    }
































    const batchStatusMatch = rawBatchStatusMatch || commandText.match(/^状態[：:\s]+([A-Za-z0-9_-]+)$/);
    if (batchStatusMatch) {
      handleBatchStatusCommand_(channel, event.thread_ts || event.ts || '', batchStatusMatch[1]);
      return;
    }




    if (commandText === '状態') {
      handleSlackStatus(channel);
      return;
    }
































    if (commandText === 'プロンプトURL' || commandText === '画像プロンプトURL' || commandText === 'Driveプロンプト') {
      notifyLatestWaitingPromptDrive_(channel, event.thread_ts || event.ts || '');

    }
































    const batchResumeMatch = commandText.match(/^バッチ再開[：:\s]+([A-Za-z0-9_-]+)(?:\s+([A-Za-z0-9_-]+))?$/);
    if (batchResumeMatch) {
      try {
        const resumeThreadTs = event.thread_ts || event.ts || '';
        const resumed = resumeZipBatchById_(batchResumeMatch[1], channel, resumeThreadTs, batchResumeMatch[2] || '');
        postSlackMessage(
          channel,
          'バッチ再開受付\nBATCH_ID：' + resumed.batchId +
          '\n未処理：' + resumed.pendingCount + '件' +
          '\n失敗から復旧：' + resumed.resetCount + '件' +
          '\n公開済み：' + resumed.publishedCount + '件' +
          '\n完了通知：元TEXT ZIPスレッドへ集約',
          resumeThreadTs
        );
      } catch (err) {
        postSlackMessage(
          channel,
          'バッチ再開エラー：' + String(err && err.message ? err.message : err),

        channel,
        event.thread_ts || event.ts || '',

      );
      return;
    }
















    if (commandText === '画像再処理' || commandText === '再処理') {

    if (driveTextRetryMatch) {
      handleDriveTextZipRecovery_(
        channel,
        event.thread_ts || event.ts || '',
        driveTextRetryMatch[1]
      );
      return;
    }
















    if (commandText === '文章再処理' || commandText === '文章ZIP再処理' || commandText === 'ZIP再処理') {
      const recovered = recoverMissedTextZipUploads_(true);
      postSlackMessage(channel, recovered ? '未受理の文章ZIPを検出して受付処理を開始しました。' : '再処理対象の未受理文章ZIPは見つかりませんでした。');

    }
































    if (commandText === 'リセット' || commandText === 'reset') {
      deletePendingArticle(getSlackUserKey(channel));
      postSlackMessage(channel, '保存中の記事データを削除しました。');
      return;
    }
































    if (commandText === '公開' || commandText === '公開:') {
      handleSlackPublishCommand(channel);
      return;
    }
































    const cleaned = normalizeArticleInput(text);
































    if (containsWrongBlogInfo(cleaned)) {
      postSlackMessage(channel, WRONG_SITE_MESSAGE);
      return;
    }
































    // .txtファイルでも長文直貼りでも、同じ形式判定で受ける
    if (isArticleText(cleaned)) {
      handleSlackArticleText(channel, cleaned);
      return;
    }
































    // 普通のテキストだけのSlackメッセージは記事入力として誤判定しない。
    // 固定タグの一部が見える場合だけ、不完全な記事入力として具体的に案内する。
    const looksLikeArticleAttempt = /【(?:TITLE|CATEGORY|META_DESCRIPTION|HTML|X_POST|IMAGE_PROMPT)_(?:START|END)】/.test(cleaned);
    if (looksLikeArticleAttempt) {
      postSlackMessage(
        channel,
        '❌ 固定タグ不足：TITLE / CATEGORY / META_DESCRIPTION / HTML / X_POST / IMAGE_PROMPT のSTART・ENDタグをすべて確認してください。',
        event.thread_ts || event.ts || ''
      );
    }
    return;
































  } catch (err) {
    console.error(err);
































    if (event && event.channel) {
      postSlackMessage(event.channel, '❌ エラー：' + err.message);
    }
  }
}

/**
 * BATCH_ID指定の公開状態確認。
 */
function handleBatchStatusCommand_(channel, threadTs, batchId) {
  batchId = String(batchId || '').trim();
  if (!batchId) {
    postSlackMessage(channel, 'BATCH_IDを指定してください。', threadTs);
    return;
  }




  let state = null;
  try { state = loadBatchStateFromLedger_(batchId); } catch (err) {}
  if (!state) {
    const stateFileId = getOptionalProp(PENDING_BATCH_PREFIX + batchId);
    if (stateFileId) {
      try { state = loadZipBatchState_(stateFileId); } catch (err) {}
    }
  }




  if (!state) {
    postSlackMessage(channel, 'バッチ状態が見つかりません。BATCH_ID：' + batchId, threadTs);
    return;
  }




  const items = Array.isArray(state.items) ? state.items : [];
  const published = items.filter(item => item.status === 'published').length;
  const pending = items.filter(item => item.status === 'pending').length;
  const failed = items.filter(item => item.status === 'failed').length;
  const duplicate = items.filter(item => item.status === 'duplicate').length;




  const lines = [
    'BATCH_ID：' + batchId,
    '状態：' + String(state.status || 'unknown'),
    '公開済み：' + published + '/' + items.length,
    '未処理：' + pending + '件',
    '失敗：' + failed + '件',
    '重複防止：' + duplicate + '件',
    '完了通知：' + (state.summarySent ? '済み' : '未送信')
  ];




  items.forEach(item => {
    const mark = item.status === 'published' ? '✅' : item.status === 'failed' ? '❌' : item.status === 'duplicate' ? '⏭️' : '⏳';
    lines.push(mark + ' ' + pad2_(item.index || item.imageIndex || 0) + ' ' + String(item.status || '') + (item.publicUrl ? ' ' + item.publicUrl : ''));
  });




  postSlackMessage(channel, lines.join('\n'), threadTs);
}

/**
 * SlackファイルがZIPか判定
 */
function isSlackZipFile(file) {
  const mimetype = String(file.mimetype || '').toLowerCase();
  const filetype = String(file.filetype || '').toLowerCase();
  const name = String(file.name || file.title || '').toLowerCase();
































  return (
    name.endsWith('.zip') ||
    filetype === 'zip' ||
    mimetype === 'application/zip' ||
    mimetype === 'application/x-zip-compressed' ||
    mimetype === 'application/octet-stream' && name.endsWith('.zip')
  );
}
































/**
 * 固定Code.gsのWebhookキューを、現在イベント処理後に少量だけ追加消化する。
 * 1分トリガー1件固定によるSlackコマンド滞留を抑えるため、追加は最大2件・120秒以内。
 */
function drainAdditionalDriveWebhookEvents_(currentBody) {
  if (
    typeof peekDriveWebhookJob_ !== 'function' ||
    typeof completeDriveWebhookJob_ !== 'function'
  ) return;




  // 外側processDriveQueue()がまだ現在jobをqueue先頭に残しているため、
  // 同一event_idなら先にqueueだけ完了させ、同じイベントを再実行しない。
  try {
    const head = peekDriveWebhookJob_();
    const currentEventId = String(currentBody && currentBody.event_id || '');
    const headEventId = String(head && head.body && head.body.event_id || '');
    if (head && currentEventId && headEventId === currentEventId) {
      completeDriveWebhookJob_(head.id);
    }
  } catch (headErr) {
    console.error('Webhookキュー先頭整理失敗：' + String(
      headErr && headErr.message ? headErr.message : headErr
    ));
  }




  const startedAt = Date.now();
  const maxExtraEvents = 2;




  for (let i = 0; i < maxExtraEvents; i++) {
    if (Date.now() - startedAt > 120000) break;




    const job = peekDriveWebhookJob_();
    if (!job || !job.body) break;




    try {
      const body = job.body;
      if (body.type === 'event_callback' && body.event) {
        const eventId = body.event_id || body.event.event_ts || body.event.ts || '';
        if (!eventId || claimSlackEvent_(eventId)) {
          try {
            handleSlackEvent(body.event);
            if (eventId) completeSlackEvent_(eventId, 'done', '');
          } catch (err) {
            console.error(err);
            if (eventId) {
              completeSlackEvent_(
                eventId,
                'failed',
                String(err && err.message ? err.message : err)
              );
            }
            const channel = body.event.channel;
            if (channel) {
              postSlackMessage(channel, '❌ Slackイベント処理エラー：' + err.message);
            }
          }
        }
      }
    } finally {
      completeDriveWebhookJob_(job.id);
    }
  }
}

/**
 * Slack履歴の補完処理を指定間隔に制限する。
 */
function shouldRunSlackRecovery_(propertyKey, force) {
  if (force) return true;




  const props = PropertiesService.getScriptProperties();
  const now = Date.now();
  const last = Number(props.getProperty(propertyKey) || 0);
  if (last && now - last < SLACK_RECOVERY_SCAN_INTERVAL_MS) return false;




  setScriptPropertySafe_(propertyKey, String(now));
  return true;
}
















/**
 * Slack Events APIで取りこぼした文章ZIPをチャンネル履歴から補完する。
 * 対象は直近6時間の未claimファイルだけ。再送せず、Slack上の既存ファイルをそのまま受理する。
 * 1回の呼出しでconversations.historyを1回、文章ZIPを最大1件だけ処理する。
 */
function recoverMissedTextZipUploads_(force) {
  try {
    if (!shouldRunSlackRecovery_(SLACK_TEXT_RECOVERY_LAST_RUN_KEY, force)) return false;




    const channel = getOptionalProp(CONFIG_KEYS.SLACK_CHANNEL_ID);
    if (!channel) return false;




    const oldest = Math.max(
      0,
      Math.floor(Date.now() / 1000) - SLACK_TEXT_RECOVERY_LOOKBACK_SECONDS
    );
    const data = slackApiGet_('conversations.history', {
      channel: channel,
      oldest: String(oldest),
      inclusive: 'true',
      limit: 100
    });
    const messages = Array.isArray(data.messages) ? data.messages : [];
    const props = PropertiesService.getScriptProperties();




    // 古い投稿から順に確認し、未受理バッチの順序を保つ。
    for (let m = messages.length - 1; m >= 0; m--) {
      const message = messages[m] || {};
      const files = Array.isArray(message.files) ? message.files : [];




      for (let f = 0; f < files.length; f++) {
        const file = files[f] || {};
        const name = String(file.name || file.title || '');
        if (!/^HIMANEKO_TEXT_BATCH_.*\.zip$/i.test(name)) continue;




        const fileId = String(file.id || '');
        if (fileId) {
          const stateKey = SLACK_FILE_STATE_PREFIX + fileId;
          const rawFileState = props.getProperty(stateKey);
          if (rawFileState) {
            const fileState = parseJsonSafe_(rawFileState, {});
            const age = Date.now() - Number(fileState.ts || 0);




            // 明示的な「文章ZIP再処理」ではfailed claimを解除して同じSlack File IDを再利用する。
            // 自動補完では従来どおりfresh failedを再処理せず、無限リトライを防ぐ。
            if (force && fileState.status === 'failed') {
              props.deleteProperty(stateKey);
            } else {
              const isFreshDoneOrFailed =
                ['done', 'failed'].indexOf(fileState.status) !== -1 &&
                age < SLACK_FILE_DONE_RETENTION_MS;
              const isFreshProcessing =
                fileState.status === 'processing' &&
                age < SLACK_FILE_PROCESSING_TTL_MS;




              // 処理中のままGAS実行が途切れた場合は、TTL経過後に再claimできるようにする。
              // 旧実装は状態キーが存在するだけで永久スキップしていたため、stale processingを救済できなかった。
              if (isFreshDoneOrFailed || isFreshProcessing) continue;
            }
          }
        }




        // 同じBATCH_IDがすでにwaiting_imagesとして受理済みなら、
        // 取りこぼした旧投稿を再処理・再通知せず、ファイル状態だけ完了へ揃える。
        const identity = parseTextBatchIdentityFromFileName_(name);
        if (identity) {
          const existingStateFileId = props.getProperty(PENDING_BATCH_PREFIX + identity.batchId);
          if (existingStateFileId) {
            try {
              const existingState = loadZipBatchState_(existingStateFileId);
              if (existingState && existingState.status === 'waiting_images') {
                completeSlackFile_(
                  file,
                  'done',
                  'reconciled_existing_batch:' + identity.batchId
                );
                continue;
              }
            } catch (err) {
              console.error('文章ZIP既存バッチ照合エラー: ' + err.message);
            }
          }
        }




        const threadTs = String(message.thread_ts || message.ts || '');
        if (handleSlackZipFileMessage(channel, file, threadTs)) {
          return true;
        }
      }
    }
  } catch (err) {
    console.error('文章ZIP補完エラー: ' + String(err && err.message ? err.message : err));
  }
  return false;
}
















/**
 * 文章ZIPファイル名からBATCH_IDと件数を取り出す。
 */
function parseTextBatchIdentityFromFileName_(fileName) {
  const match = String(fileName || '').match(
    /^HIMANEKO_TEXT_BATCH_(.+)_(\d+)\.zip$/i
  );
  if (!match) return null;




  return {
    batchId: String(match[1] || ''),
    total: Number(match[2] || 0)
  };
}
















/**
 * Slackファイルイベントが未着だったwaiting_imagesバッチを補完する。
 * 5分間隔で、待機バッチをローテーションし、1回の呼出しでSlack APIを最大1回だけ使う。
 */
function recoverWaitingImageBatches_(force) {
  try {
    if (!shouldRunSlackRecovery_(SLACK_IMAGE_RECOVERY_LAST_RUN_KEY, force)) return false;




    const props = PropertiesService.getScriptProperties();
    const all = props.getProperties();
    const pendingKeys = Object.keys(all)
      .filter(key => key.startsWith(PENDING_BATCH_PREFIX))
      .sort();
    if (!pendingKeys.length) return false;




    const startIndex = Number(props.getProperty(SLACK_IMAGE_RECOVERY_CURSOR_KEY) || 0) % pendingKeys.length;




    for (let offset = 0; offset < pendingKeys.length; offset++) {
      const index = (startIndex + offset) % pendingKeys.length;
      const stateFileId = all[pendingKeys[index]];
      if (!stateFileId) continue;




      let state;
      try {
        state = loadZipBatchState_(stateFileId);
      } catch (err) {
        console.error(err);
        continue;
      }




      if (!state || state.status !== 'waiting_images' || !state.channel || !state.threadTs) continue;




      setScriptPropertySafe_(
        SLACK_IMAGE_RECOVERY_CURSOR_KEY,
        String((index + 1) % pendingKeys.length)
      );




      const data = slackApiGet_('conversations.replies', {
        channel: state.channel,
        ts: state.threadTs,
        limit: 100
      });
      const messages = Array.isArray(data.messages) ? data.messages : [];
      const expectedPrefix = 'HIMANEKO_IMAGE_BATCH_' + state.batchId + '_';




      for (let m = messages.length - 1; m >= 0; m--) {
        const files = Array.isArray(messages[m].files) ? messages[m].files : [];
        for (let f = 0; f < files.length; f++) {
          const file = files[f];
          const name = String(file.name || file.title || '');
          if (!name.startsWith(expectedPrefix) || !/\.zip$/i.test(name)) continue;




          const sameRecoveredFile = Boolean(
            state.lastRecoveredImageFileId &&
            state.lastRecoveredImageFileId === file.id
          );
          if (sameRecoveredFile && !force) continue;




          if (force && file.id) {
            const fileStateKey = SLACK_FILE_STATE_PREFIX + file.id;
            const fileState = parseJsonSafe_(props.getProperty(fileStateKey), {});
            if (fileState.status === 'failed') {
              props.deleteProperty(fileStateKey);
            } else if (fileState.status === 'done') {
              state.lastRecoveredImageFileId = file.id || '';
              state.lastRecoveryCheckedAt = new Date().toISOString();
              state.lastRecoveryError = '';
              saveZipBatchState_(state);
              continue;
            }
          }




          state.lastRecoveryCheckedAt = new Date().toISOString();
          saveZipBatchState_(state);
          const recovered = handleSlackZipFileMessage(state.channel, file, state.threadTs);
          if (recovered) {
            state.lastRecoveredImageFileId = file.id || '';
            state.lastRecoveryError = '';
            state.lastRecoveryCheckedAt = new Date().toISOString();
            saveZipBatchState_(state);
            return true;
          }




          state.lastRecoveryError = '画像ZIP再処理失敗 file_id=' + String(file.id || '');
          state.lastRecoveryCheckedAt = new Date().toISOString();
          saveZipBatchState_(state);
          return false;
        }
      }




      state.lastRecoveryCheckedAt = new Date().toISOString();
      saveZipBatchState_(state);
      return false;
    }
  } catch (err) {
    console.error('waiting_images補完エラー: ' + String(err && err.message ? err.message : err));
  }
  return false;
}

/**
 * Slack Web APIのGET共通処理。
 */
function slackApiGet_(method, params) {
  const token = getProp(CONFIG_KEYS.SLACK_BOT_TOKEN);
  const query = Object.keys(params || {})
    .map(key => encodeURIComponent(key) + '=' + encodeURIComponent(String(params[key])))
    .join('&');
  const res = UrlFetchApp.fetch('https://slack.com/api/' + method + (query ? '?' + query : ''), {
    method: 'get',
    headers: { Authorization: 'Bearer ' + token },
    muteHttpExceptions: true
  });
  const body = res.getContentText();
  let data = {};
  try { data = JSON.parse(body); } catch (err) {}
  if (res.getResponseCode() < 200 || res.getResponseCode() >= 300 || !data.ok) {
    throw new Error('Slack API ' + method + ' 失敗：' + body);
  }
  return data;
}
































/**
 * Slackへ送られたZIPを展開し、1〜10件の可変バッチとしてキューへ登録する。
 */
function handleSlackZipFileMessage(channel, file, threadTs, preloadedZipBlob) {
  // 「検査を開始します」を出す前にSlack file_idをclaimする。
  // これによりWebhook再送とwaiting_images補完の同時処理を止める。
  if (!claimSlackFile_(file)) return false;
















  let completed = false;
  let textBatchId = '';
  let textBatchClaimed = false;
  try {
    const incomingName = String(file.name || file.title || 'batch.zip');




    // 正規文章ZIPはダウンロード前にBATCH_IDで排他する。
    // Slackが同名ファイルを別File IDで再送しても、成功済み/処理中バッチを再取得しない。
    const textIdentity = parseTextBatchIdentityFromFileName_(incomingName);
    if (textIdentity) {
      textBatchId = textIdentity.batchId;




      const existingStateFileId = getOptionalProp(PENDING_BATCH_PREFIX + textBatchId);
      if (existingStateFileId) {
        try {
          const existingState = loadZipBatchState_(existingStateFileId);
          if (existingState && existingState.status === 'waiting_images') {
            completeTextBatchAcceptance_(textBatchId, 'done', String(file.id || ''), 'reconciled_existing_batch');
            completed = true;
            return true;
          }
        } catch (stateErr) {
          console.error('文章ZIP既存バッチ事前照合エラー: ' + stateErr.message);
        }
      }




      if (!claimTextBatchAcceptance_(textBatchId, String(file.id || ''))) {
        completed = true;
        return true;
      }
      textBatchClaimed = true;
    }
















    // 旧方式の画像プロンプトZIPは入力として再解析しない。現行方式はDriveフォルダURL通知。
    if (/^HIMANEKO_IMAGE_PROMPTS_/i.test(incomingName)) {
      postSlackMessage(channel, '画像プロンプトZIPは旧方式の返却物のため受付対象外です。現行方式ではDriveフォルダURLを使用します。', threadTs);
      completed = true;
      return true;
    }
















    postSlackMessage(channel, 'ZIPを受信しました。ZIP構造と記事対応を確認します：' + incomingName, threadTs);
















    const declaredSize = Number(file.size || 0);
    if (declaredSize > ZIP_BATCH_MAX_BYTES) {
      throw new Error('ZIPは40MB以下にしてください。');
    }
















    const zipBlob = preloadedZipBlob || getSlackFileBlob(file);
    const actualSize = zipBlob.getBytes().length;
    if (actualSize > ZIP_BATCH_MAX_BYTES) {
      throw new Error('ZIPは40MB以下にしてください。');
    }
















    let unzipped;
    try {
      unzipped = Utilities.unzip(zipBlob);
    } catch (err) {
      throw new Error('ZIPを展開できませんでした。通常のZIP形式で作り直してください。');
    }
















    const zipName = String(file.name || file.title || zipBlob.getName() || 'batch.zip');
    const cleanEntries = getCleanZipEntries_(unzipped);




    // 新規ZIPは正規名・正規構造だけを受け付ける。
    if (/^HIMANEKO_TEXT_BATCH_/i.test(zipName)) {
      handleTextBatchZip_(channel, threadTs, file, zipBlob, cleanEntries);
      completed = true;
      return true;
    }
    if (/^HIMANEKO_IMAGE_BATCH_/i.test(zipName)) {
      handleImageBatchZip_(channel, threadTs, file, zipBlob, cleanEntries);
      completed = true;
      return true;
    }




    // 旧形式は、選択済みまたは画像待機中の既存バッチと全件一致する場合だけ救済する。
    const legacyEntries = stripOneCommonZipRootEntries_(cleanEntries);
    const txtCount = legacyEntries.filter(entry => /\.txt$/i.test(entry.basename)).length;
    const imageCount = legacyEntries.filter(entry => /\.png$/i.test(entry.basename)).length;
    if (txtCount > 0 && imageCount === 0) {
      validateLegacyTextZipEntries_(legacyEntries);
      handleTextBatchZip_(channel, threadTs, file, zipBlob, legacyEntries);
      completed = true;
      return true;
    }
    if (imageCount > 0 && txtCount === 0) {
      const legacyContext = resolveLegacyImageZipContext_(legacyEntries);
      handleImageBatchZip_(channel, threadTs, file, zipBlob, legacyEntries, legacyContext);
      completed = true;
      return true;
    }




    throw new Error(
      'ZIP種別または旧形式の照合対象を確認できません。新規ZIPは正規名・正規構造、旧形式は既存バッチとの全件一致が必要です。ZIP名：' + zipName
    );
  } catch (err) {
    console.error(err);
    if (textBatchClaimed && textBatchId) {
      completeTextBatchAcceptance_(textBatchId, 'release', String(file.id || ''), String(err && err.message ? err.message : err));
    }
    postSlackMessage(channel, '❌ ZIP受付エラー：' + err.message, threadTs);
    completeSlackFile_(file, 'failed', String(err && err.message ? err.message : err));
    return false;
  } finally {
    if (completed) {
      if (textBatchId) {
        completeTextBatchAcceptance_(textBatchId, 'done', String(file.id || ''), '');
      }
      completeSlackFile_(file, 'done', '');
      cleanupOldSlackEventKeys_();
    }
  }
}
















/**
 * DriveAppの一時障害を、重複を作らない形で再試行する。
 * Service error: Drive 等だけを対象にし、入力不備や権限エラーは即時に投げ直す。
 */
function isRetryableDriveError_(err) {
  const message = String(err && err.message ? err.message : err || '');
  return /Service error:\s*Drive|Drive service|backendError|internalError|Internal error|Rate Limit|rateLimitExceeded|userRateLimitExceeded|temporar|try again|timeout|timed out/i.test(message);
}




function withDriveRetry_(label, operation, maxAttempts) {
  const attempts = Math.max(1, Math.min(Number(maxAttempts || DRIVE_OPERATION_MAX_ATTEMPTS), 6));
  let lastError = null;
  for (let attempt = 1; attempt <= attempts; attempt++) {
    try {
      return operation(attempt);
    } catch (err) {
      lastError = err;
      if (!isRetryableDriveError_(err) || attempt >= attempts) throw err;
      const waitMs = DRIVE_OPERATION_RETRY_BASE_MS * Math.pow(2, attempt - 1);
      console.error('Drive一時エラー再試行：' + label + ' / ' + attempt + '/' + attempts + ' / ' + String(err && err.message ? err.message : err));
      Utilities.sleep(waitMs);
    }
  }
  throw lastError || new Error('Drive再試行に失敗しました：' + label);
}




function getOrCreateChildFolderWithDriveRetry_(parentFolder, folderName) {
  return withDriveRetry_('フォルダ取得/作成 ' + folderName, function() {
    const matches = parentFolder.getFoldersByName(folderName);
    if (matches.hasNext()) return matches.next();
    return parentFolder.createFolder(folderName);
  });
}




function getOrCreateBlobFileWithDriveRetry_(folder, fileName, blobFactory) {
  return withDriveRetry_('ファイル取得/作成 ' + fileName, function() {
    const existing = folder.getFilesByName(fileName);
    if (existing.hasNext()) return existing.next();
    const blob = blobFactory();
    if (!blob) throw new Error('Drive作成用Blobがありません：' + fileName);
    blob.setName(fileName);
    return folder.createFile(blob);
  });
}




function getOrCreateTextFileWithDriveRetry_(folder, fileName, text) {
  return withDriveRetry_('テキスト取得/作成 ' + fileName, function() {
    const existing = folder.getFilesByName(fileName);
    if (existing.hasNext()) return existing.next();
    return folder.createFile(fileName, String(text || ''), MimeType.PLAIN_TEXT);
  });
}




function handleTextBatchZip_(channel, threadTs, file, zipBlob, cleanEntries) {
  const zipHash = computeBlobSha256_(zipBlob);
  const duplicateRaw = getOptionalProp(ZIP_BATCH_HASH_PREFIX + zipHash);
  if (duplicateRaw) {
    postSlackMessage(channel, 'この文章ZIPはすでに受付済みです。重複投稿を防止しました。再送せず、既存スレッドの返信を確認してください。', threadTs);
    return;
  }
































  const manifest = buildTextZipManifest_(cleanEntries);
  if (manifest.errors.length) {
    throw new Error(manifest.errors.join(' / '));
  }
  if (manifest.total < 1 || manifest.total > ZIP_BATCH_MAX_ARTICLES) {
    throw new Error('文章ZIPの記事数は1〜10件にしてください。現在：' + manifest.total + '件');
  }
































  const batchId = manifest.batchId;
  const pendingKey = PENDING_BATCH_PREFIX + batchId;
  const existingStateFileId = getOptionalProp(pendingKey);
  if (existingStateFileId) {
    throw new Error('同じBATCH_IDの保留バッチがすでに存在します：' + batchId);
  }
































  const queueRoot = getOrCreateZipBatchQueueFolder_();




  // batch-state.json作成前にGASが強制終了すると、同名の途中フォルダだけが残る。
  // pendingKeyが存在しない状態で同BATCHを再受付する場合、その孤児フォルダだけを退避して作り直す。
  const batchFolderName = sanitizeDriveName_(batchId);
  const sameNameFolders = queueRoot.getFoldersByName(batchFolderName);
  while (sameNameFolders.hasNext()) {
    const candidateFolder = sameNameFolders.next();
    const stateFiles = candidateFolder.getFilesByName('batch-state.json');
    if (!stateFiles.hasNext()) {
      try {
        candidateFolder.setTrashed(true);
      } catch (cleanupErr) {
        console.error('孤児バッチフォルダ退避失敗：' + cleanupErr.message);
      }
    }
  }




  const batchFolder = getOrCreateChildFolderWithDriveRetry_(queueRoot, batchFolderName);
  const items = [];
































  manifest.items.forEach(sourceItem => {
    const itemFolderName = pad2_(sourceItem.imageIndex) + '_' + sanitizeDriveName_(sourceItem.workId);
    const itemFolder = getOrCreateChildFolderWithDriveRetry_(batchFolder, itemFolderName);
    const txtFileName = String(sourceItem.txtBlob.getName() || 'article.txt');
    const txtFile = getOrCreateBlobFileWithDriveRetry_(itemFolder, txtFileName, function() {
      return sourceItem.txtBlob.copyBlob ? sourceItem.txtBlob.copyBlob() : sourceItem.txtBlob;
    });
    const promptFile = getOrCreateBlobFileWithDriveRetry_(itemFolder, sourceItem.promptFileName, function() {
      return Utilities.newBlob(
        sourceItem.promptFileText,
        MimeType.PLAIN_TEXT,
        sourceItem.promptFileName
      );
    });
































    items.push({
      index: sourceItem.imageIndex,
      imageIndex: sourceItem.imageIndex,
      batchId: sourceItem.batchId,
      workId: sourceItem.workId,
      packageFolderName: sourceItem.packageFolderName,
      expectedImageFileName: sourceItem.imageFileName,
      promptFileName: sourceItem.promptFileName,
      displayName: sourceItem.packageFolderName,
      status: 'waiting_image',
      stage: 'drive_prompt_ready',
      attempts: 0,
      validationError: '',
      error: '',
      itemFolderId: itemFolder.getId(),
      txtFileId: txtFile.getId(),
      promptFileId: promptFile.getId(),
      imageFileId: '',
      imageUrl: '',
      editUrl: '',
      previewUrl: '',
      publicUrl: '',
      articleHash: sourceItem.articleHash,
      title: sourceItem.title
    });
  });
































  const state = {
    schemaVersion: 2,
    pipeline: 'EMBEDDED_SLOT_HANDOFF',
    batchId,
    zipHash,
    zipName: String(file.name || file.title || 'articles.zip'),
    channel,
    threadTs: threadTs || '',
    batchFolderId: batchFolder.getId(),
    stateFileId: '',
    createdAt: new Date().toISOString(),
    updatedAt: new Date().toISOString(),
    status: 'waiting_images',
    total: items.length,
    items,
    summarySent: false,
    promptFolderId: '',
    promptFolderDriveUrl: '',
    promptManifestFileId: '',
    promptManifestDriveUrl: '',
    promptNotificationAttempts: 0,
    promptNotificationStatus: 'pending',
    promptNotificationError: ''
  };
































  const stateFile = getOrCreateTextFileWithDriveRetry_(batchFolder, 'batch-state.json', JSON.stringify(state));
  state.stateFileId = stateFile.getId();
  saveZipBatchState_(state);
  safeSyncBatchStateToArticleLedger_(state);
































  const props = PropertiesService.getScriptProperties();
  setScriptPropertySafe_(pendingKey, state.stateFileId);
  setScriptPropertySafe_(
    ZIP_BATCH_HASH_PREFIX + zipHash,
    JSON.stringify({ stateFileId: state.stateFileId, status: state.status, ts: Date.now() })
  );
  cleanupOldZipBatchHashKeys_();
































  ensureImagePromptDrivePackageForState_(state);
  const notified = postSlackMessageWithRetry_(
    channel,
    buildImagePromptDriveSlackMessage_(state, false),
    threadTs,
    CRITICAL_SLACK_MESSAGE_MAX_ATTEMPTS
  );
































  state.promptNotificationAttempts = notified ? 1 : CRITICAL_SLACK_MESSAGE_MAX_ATTEMPTS;
  state.promptNotificationStatus = notified ? 'sent' : 'failed';
  state.promptNotificationError = notified ? '' : 'Slack URL通知が合計2回失敗したため停止';
  saveZipBatchState_(state);
  safeSyncBatchStateToArticleLedger_(state);
}
































function handleImageBatchZip_(channel, threadTs, file, zipBlob, cleanEntries, legacyContext) {
  const zipName = String(file.name || file.title || zipBlob.getName() || 'images.zip');
  const nameMatch = zipName.match(/^HIMANEKO_IMAGE_BATCH_(.+)_(10|[1-9])\.zip$/i);
  if (!nameMatch && !legacyContext) {
    throw new Error('画像ZIP名が不正です。HIMANEKO_IMAGE_BATCH_{BATCH_ID}_{1〜10の件数}.zip にしてください。');
  }
















  const batchId = legacyContext ? legacyContext.batchId : nameMatch[1];
  const declaredTotal = legacyContext ? Number(legacyContext.total) : Number(nameMatch[2]);
  const slackFileId = String(file && file.id || '');
















  if (!claimImageBatchAcceptance_(batchId, slackFileId)) {
    postSlackMessage(
      channel,
      'このBATCH_IDの画像ZIPはすでに受付済み、または処理中です。再送せず現在の処理結果を確認してください。BATCH_ID：' + batchId,
      threadTs
    );
    return false;
  }
















  let imageBatchAccepted = false;
  try {
    const stateFileId = getOptionalProp(PENDING_BATCH_PREFIX + batchId);
    if (!stateFileId) {
      throw new Error('対応する保留中の文章バッチが見つかりません。BATCH_ID：' + batchId);
    }
































    const state = loadZipBatchState_(stateFileId);
    if (state.status !== 'waiting_images') {
      throw new Error('このバッチは画像受付待ちではありません。現在状態：' + state.status);
    }
    if (declaredTotal !== state.total) {
      throw new Error('画像ZIP名の件数と文章バッチ件数が一致しません。ZIP：' + declaredTotal + '、文章：' + state.total);
    }
































    const zipHash = computeBlobSha256_(zipBlob);
    if (getOptionalProp(IMAGE_ZIP_HASH_PREFIX + zipHash)) {
      postSlackMessage(channel, 'この画像ZIPはすでに受付済みです。重複投稿を防止しました。', threadTs || state.threadTs);
      return;
    }
































    const manifest = buildImageZipManifest_(cleanEntries);
    if (manifest.errors.length) {
      throw new Error(manifest.errors.join(' / '));
    }
    if (manifest.total !== state.total) {
      throw new Error('画像数が一致しません。必要：' + state.total + '枚、実際：' + manifest.total + '枚');
    }
































    const byName = {};
    manifest.items.forEach(item => { byName[item.fileName] = item; });
    const missing = [];
    const expected = {};
    state.items.forEach(item => {
      expected[item.expectedImageFileName] = true;
      if (!byName[item.expectedImageFileName]) missing.push(item.expectedImageFileName);
    });
    const extras = manifest.items.filter(item => !expected[item.fileName]).map(item => item.fileName);
    if (missing.length || extras.length) {
      throw new Error(
        '画像ファイル名が一致しません。' +
        (missing.length ? '不足：' + missing.join(', ') + '。' : '') +
        (extras.length ? '余分：' + extras.join(', ') + '。' : '')
      );
    }
































    state.items.forEach(item => {
      const source = byName[item.expectedImageFileName];
      const itemFolder = DriveApp.getFolderById(item.itemFolderId);
      const saved = itemFolder.createFile(source.blob.copyBlob().setName(item.expectedImageFileName));
      item.imageFileId = saved.getId();
      item.status = 'pending';
      item.stage = 'image_received';
      item.error = '';
    });
































    state.status = 'queued';
    state.imageZipName = zipName;
    state.imageZipHash = zipHash;
    state.imageZipReceivedAt = new Date().toISOString();
    saveZipBatchState_(state);
    safeSyncBatchStateToArticleLedger_(state);
    enqueueZipBatchState_(state.stateFileId);
































    setScriptPropertySafe_(
      IMAGE_ZIP_HASH_PREFIX + zipHash,
      JSON.stringify({ stateFileId: state.stateFileId, batchId, status: 'queued', ts: Date.now() })
    );
    cleanupOldZipBatchHashKeys_();
































    postSlackMessage(
      channel,
      '画像ZIPを受け付けました。\nBATCH_ID：' + batchId + '\n画像：' + state.total + '枚\n照合PASS。順番にブログ公開を開始します。',
      threadTs || state.threadTs
    );
















    imageBatchAccepted = true;
    return true;
  } finally {
    completeImageBatchAcceptance_(
      batchId,
      imageBatchAccepted ? 'done' : 'release',
      slackFileId,
      imageBatchAccepted ? '' : '画像ZIP受付処理が完了しませんでした'
    );
  }
}
















function ensureImagePromptDrivePackageForState_(state) {
  if (!state || !state.batchFolderId || !state.items || !state.items.length) {
    throw new Error('画像プロンプトDriveパッケージの作成情報が不足しています');
  }
































  if (state.promptFolderId) {
    try {
      const existingFolder = withDriveRetry_('既存プロンプトフォルダ取得 ' + state.batchId, function() {
        return DriveApp.getFolderById(state.promptFolderId);
      });
      state.promptFolderDriveUrl = existingFolder.getUrl();
      if (state.promptManifestFileId) {
        state.promptManifestDriveUrl = withDriveRetry_('既存manifest取得 ' + state.batchId, function() {
          return DriveApp.getFileById(state.promptManifestFileId).getUrl();
        });
      }
      saveZipBatchState_(state);
      return state;
    } catch (err) {
      state.promptFolderId = '';
      state.promptFolderDriveUrl = '';
      state.promptManifestFileId = '';
      state.promptManifestDriveUrl = '';
    }
  }
































  const batchFolder = withDriveRetry_('バッチフォルダ取得 ' + state.batchId, function() {
    return DriveApp.getFolderById(state.batchFolderId);
  });
  const folderName = 'HIMANEKO_IMAGE_PROMPTS_' + state.batchId + '_' + state.total;
  const promptFolder = getOrCreateChildFolderWithDriveRetry_(batchFolder, folderName);
































  const manifestItems = [];
  state.items
    .slice()
    .sort((a, b) => a.imageIndex - b.imageIndex)
    .forEach(item => {
      const sourceBlob = withDriveRetry_('元プロンプト取得 ' + item.promptFileName, function() {
        return DriveApp.getFileById(item.promptFileId).getBlob().setName(item.promptFileName);
      });
      const promptFile = getOrCreateBlobFileWithDriveRetry_(promptFolder, item.promptFileName, function() {
        return sourceBlob.copyBlob ? sourceBlob.copyBlob() : sourceBlob;
      });
      const promptText = sourceBlob.getDataAsString('UTF-8');
      const promptBytes = sourceBlob.getBytes().length;
      const promptSha256 = computeTextSha256_(promptText);
































      item.drivePromptFileId = promptFile.getId();
      item.drivePromptFileUrl = promptFile.getUrl();
      item.promptTextBytes = promptBytes;
      item.promptTextSha256 = promptSha256;
































      manifestItems.push({
        imageIndex: item.imageIndex,
        workId: item.workId,
        promptFileName: item.promptFileName,
        imageFileName: item.expectedImageFileName,
        promptFileId: item.drivePromptFileId,
        promptFileUrl: item.drivePromptFileUrl,
        size: promptBytes,
        sha256: promptSha256
      });
    });
































  const manifestText = buildImagePromptManifestText_(state.batchId, state.total, manifestItems);
  const manifestFile = getOrCreateTextFileWithDriveRetry_(
    promptFolder,
    'manifest.txt',
    manifestText
  );
































  state.promptFolderId = promptFolder.getId();
  state.promptFolderDriveUrl = promptFolder.getUrl();
  state.promptManifestFileId = manifestFile.getId();
  state.promptManifestDriveUrl = manifestFile.getUrl();
  state.promptDriveReadyAt = new Date().toISOString();
  saveZipBatchState_(state);
  return state;
}

  ];


  (items || []).forEach(item => {
    const index = pad2_(item.imageIndex);
    lines.push(index + '_FILE=' + item.promptFileName);
    lines.push(index + '_PROMPT_FILE_ID=' + item.promptFileId);
    lines.push(index + '_PROMPT_FILE_URL=' + item.promptFileUrl);
    lines.push(index + '_WORK_ID=' + item.workId);
    lines.push(index + '_IMAGE_FILE_NAME=' + item.imageFileName);
    lines.push(index + '_SIZE=' + item.size);
    lines.push(index + '_SHA256=' + item.sha256);
    lines.push('');
  });


  return lines.join('\n');
}




function readEmbeddedPromptField_(text, key) {
  const escaped = String(key || '').replace(/[.*+?^${}()|[\]\\]/g, '\\$&');
  const match = String(text || '').match(new RegExp('^' + escaped + ':\\s*(.*)$', 'm'));
  return match ? String(match[1] || '').trim() : '';
}




function extractEmbeddedPromptBody_(text) {
  return String(text || '')
    .replace(/^\uFEFF/, '')
    .replace(/^\s*【IMAGE_PROMPT_START】\s*/m, '')
    .replace(/\s*【IMAGE_PROMPT_END】\s*$/m, '')
    .trim();
}




function buildEmbeddedImageSlot_(state, item, slotIndex) {
  const batchId = String(state && state.batchId || '').trim();
  const total = Number(state && state.total);
  const expectedIndex = Number(slotIndex);
  const fileId = String(item && (item.promptFileId || item.drivePromptFileId) || '').trim();
  if (!fileId) throw new Error('SLOT_' + pad2_(expectedIndex) + ' のIMAGE_PROMPT実体がありません。');


  const raw = withDriveRetry_('埋め込みSLOT元プロンプト取得 ' + pad2_(expectedIndex), function() {
    return DriveApp.getFileById(fileId).getBlob().getDataAsString('UTF-8');
  });
  const body = extractEmbeddedPromptBody_(raw);
  const promptBatchId = readEmbeddedPromptField_(body, 'BATCH_ID');
  const promptTotal = Number(readEmbeddedPromptField_(body, 'BATCH_TOTAL'));
  const promptIndex = Number(readEmbeddedPromptField_(body, 'IMAGE_INDEX'));
  const workId = readEmbeddedPromptField_(body, 'WORK_ID');
  const imageFileName = readEmbeddedPromptField_(body, 'IMAGE_FILE_NAME');
  const title = readEmbeddedPromptField_(body, 'TITLE');
  const titleLayoutLines = Number(readEmbeddedPromptField_(body, 'TITLE_LAYOUT_LINES'));


  if (promptBatchId !== batchId) throw new Error('SLOT_' + pad2_(expectedIndex) + ' のBATCH_IDが不一致です。');
  if (promptTotal !== total) throw new Error('SLOT_' + pad2_(expectedIndex) + ' のBATCH_TOTALが不一致です。');
  if (promptIndex !== expectedIndex) throw new Error('SLOT_' + pad2_(expectedIndex) + ' のIMAGE_INDEXが不一致です。');
  if (workId !== String(item.workId || '')) throw new Error('SLOT_' + pad2_(expectedIndex) + ' のWORK_IDが不一致です。');
  if (imageFileName !== String(item.expectedImageFileName || '')) throw new Error('SLOT_' + pad2_(expectedIndex) + ' のIMAGE_FILE_NAMEが不一致です。');
  if (!title) throw new Error('SLOT_' + pad2_(expectedIndex) + ' のTITLEがありません。');
  if (!Number.isInteger(titleLayoutLines) || titleLayoutLines < 1 || titleLayoutLines > 4) {
    throw new Error('SLOT_' + pad2_(expectedIndex) + ' のTITLE_LAYOUT_LINESが不正です。');
  }


  const titleLines = [];
  for (let i = 1; i <= titleLayoutLines; i++) {
    const line = readEmbeddedPromptField_(body, 'TITLE_LINE_' + i);
    if (!line) throw new Error('SLOT_' + pad2_(expectedIndex) + ' のTITLE_LINE_' + i + ' がありません。');
    titleLines.push(line);
  }
  const normalizeTitle = value => String(value || '').replace(/\s+/g, '');
  if (normalizeTitle(titleLines.join('')) !== normalizeTitle(title)) {
    throw new Error('SLOT_' + pad2_(expectedIndex) + ' のTITLE_LINE連結がTITLEと一致しません。');
  }


  const fields = {
    mainHook: readEmbeddedPromptField_(body, 'MAIN_HOOK'),
    supportCallouts: readEmbeddedPromptField_(body, 'SUPPORT_CALLOUTS'),
    targetName: readEmbeddedPromptField_(body, 'TARGET_NAME'),
    keyVisual: readEmbeddedPromptField_(body, 'KEY_VISUAL'),
    composition: readEmbeddedPromptField_(body, 'COMPOSITION'),
    color: readEmbeddedPromptField_(body, 'COLOR_DIRECTION'),
    energy: readEmbeddedPromptField_(body, 'ENERGY_LEVEL'),
    allowed: readEmbeddedPromptField_(body, 'ALLOWED_ELEMENTS'),
    forbidden: readEmbeddedPromptField_(body, 'FORBIDDEN_ELEMENTS')
  };
  Object.keys(fields).forEach(key => {
    if (!fields[key]) throw new Error('SLOT_' + pad2_(expectedIndex) + ' の画像設計項目が不足しています：' + key);
  });


  const lines = ['■SLOT_' + pad2_(expectedIndex), 'TITLE:'];
  titleLines.forEach((line, idx) => lines.push((idx + 1) + '段目: ' + line));
  lines.push('MAIN_HOOK: ' + fields.mainHook);
  lines.push('SUPPORT_CALLOUTS: ' + fields.supportCallouts);
  lines.push('TARGET_NAME: ' + fields.targetName);
  lines.push('KEY_VISUAL: ' + fields.keyVisual);
  lines.push('COMPOSITION: ' + fields.composition);
  lines.push('COLOR: ' + fields.color);
  lines.push('ENERGY: ' + fields.energy);
  lines.push('使う要素: ' + fields.allowed);
  lines.push('禁止: ' + fields.forbidden);
  return lines.join('\n');
}

function postSlackMessageWithRetry_(channel, message, threadTs, maxAttempts) {
  const limit = Math.max(1, Math.min(Number(maxAttempts || 1), 2));
  let lastError = '';
































  for (let attempt = 1; attempt <= limit; attempt++) {
    try {
      const ok = postSlackMessage(channel, message, threadTs);
      if (ok) return true;
      lastError = 'chat.postMessageが失敗しました';
    } catch (err) {
      lastError = String(err && err.message ? err.message : err);
    }
































    if (attempt < limit) Utilities.sleep(1000);
  }
































  console.error('Slack重要通知を合計' + limit + '回試行後に停止：' + lastError);
  return false;
}

/**
 * ZIP内ファイルを記事単位へまとめる。
 * 共通の最上位フォルダがある場合は自動的に取り除く。
 */
function getCleanZipEntries_(blobs) {
  return (blobs || [])
    .map((blob, index) => {
      const path = normalizeZipPath_(blob.getName());
      const parts = path.split('/').filter(Boolean);
      return {
        blob,
        index,
        path,
        parts,
        basename: parts[parts.length - 1] || ''
      };
    })
    .filter(entry => entry.path)
    .filter(entry => !entry.path.endsWith('/'))
    .filter(entry => !/(^|\/)__MACOSX(\/|$)/i.test(entry.path))
    .filter(entry => !/(^|\/)\.DS_Store$/i.test(entry.path))
    .filter(entry => !/(^|\/)Thumbs\.db$/i.test(entry.path));
}




function stripOneCommonZipRootEntries_(entries) {
  const list = (entries || []).slice();
  if (!list.length || list.some(entry => !entry.parts || entry.parts.length < 2)) return list;
  const root = list[0].parts[0];
  if (!root || !list.every(entry => entry.parts[0] === root)) return list;
  return list.map(entry => {
    const parts = entry.parts.slice(1);
    return Object.assign({}, entry, { parts: parts, path: parts.join('/'), basename: parts[parts.length - 1] || '' });
  });
}

function validateLegacyTextZipEntries_(entries) {
  const manifest = buildTextZipManifest_(entries);
  if (manifest.errors.length) throw new Error(manifest.errors.join(' / '));
  const loaded = readCandidateCannibalRows_(), h = candidateCannibalHeaderIndex_();
  const rows = loaded.rows.filter(row =>
    String(row[h.BATCH_ID] || '') === manifest.batchId &&
    ['SELECTED','HOLD_SELECTION_SET','HOLD_RESOURCE_SET','HANDED_TO_H1'].indexOf(String(row[h.STATUS] || '')) !== -1
  );
  const expected = rows.map(row => String(row[h.WORK_ID] || '')).filter(Boolean).sort();
  const actual = manifest.items.map(item => String(item.workId || '')).filter(Boolean).sort();
  if (!expected.length || expected.length !== actual.length || expected.join('|') !== actual.join('|')) {
    throw new Error('旧形式文章ZIPは候補・カニバリ台帳の既存選択バッチとWORK_ID全件一致が必要です。BATCH_ID：' + manifest.batchId);
  }
  return manifest;
}

  });

function buildTextZipManifest_(entries) {
  const errors = [];
  const groups = {};
































  (entries || []).forEach(entry => {
    if (!/\.txt$/i.test(entry.basename)) {
      errors.push('文章ZIPにtxt以外が入っています：' + entry.path);
      return;
    }
    if (entry.parts.length !== 2) {
      errors.push('文章ZIPは直下の記事フォルダ/article.txt構造にしてください：' + entry.path);
      return;
    }
    if (entry.basename !== 'article.txt') {
      errors.push('記事ファイル名はarticle.txt固定です：' + entry.path);
      return;
    }
    const folderName = entry.parts[0];
    if (!groups[folderName]) groups[folderName] = [];
    groups[folderName].push(entry);
  });
































  const items = [];
  Object.keys(groups).sort(naturalCompare_).forEach(folderName => {
    const group = groups[folderName];
    if (group.length !== 1) {
      errors.push(folderName + '内のarticle.txtは1個だけ必要です。');
      return;
    }
































    try {
      const txtBlob = group[0].blob.copyBlob().setName('article.txt');
      const articleText = txtBlob.getDataAsString('UTF-8').replace(/^\uFEFF/, '');
      const cleaned = normalizeArticleInput(articleText);
      if (containsWrongBlogInfo(cleaned)) throw new Error('別ブログ情報が混在しています');
      if (!isArticleText(cleaned)) throw new Error('必要な固定タグが不足しています');
































      const parsed = parseArticleOutput(cleaned);
      if (!parsed.ok) throw new Error(parsed.errors.join(' / '));
































      const meta = parseImagePromptMetadata_(cleaned);
      if (meta.packageFolderName !== folderName) {
        throw new Error('PACKAGE_FOLDER_NAMEと記事フォルダ名が一致しません');
      }
      if (meta.title !== parsed.title) {
        throw new Error('IMAGE_PROMPT内TITLEと記事TITLEが一致しません');
      }
































      items.push({
        batchId: meta.batchId,
        imageIndex: meta.imageIndex,
        workId: meta.workId,
        packageFolderName: meta.packageFolderName,
        imageFileName: meta.imageFileName,
        title: meta.title,
        seasonJp: meta.seasonJp,
        promptFileName: pad2_(meta.imageIndex) + '_' + meta.workId + '_image_prompt.txt',
        promptFileText: '【IMAGE_PROMPT_START】\n' + meta.imagePrompt.trim() + '\n【IMAGE_PROMPT_END】\n',
        txtBlob,
        articleHash: computeTextSha256_(cleaned)
      });
    } catch (err) {
      errors.push(folderName + '：' + err.message);
    }
  });
































  const batchIds = uniqueArray(items.map(item => item.batchId));
  // 記事解析が全件失敗した場合はBATCH_ID不一致を二次エラーとして追加しない。
  // 1件以上解析できた場合だけ、実際に複数BATCH_IDが混在しているかを検査する。
  if (items.length > 0 && batchIds.length !== 1) errors.push('BATCH_IDは全記事で同一にしてください。');
  const workIds = items.map(item => item.workId);
  if (uniqueArray(workIds).length !== workIds.length) errors.push('WORK_IDが重複しています。');
  const imageNames = items.map(item => item.imageFileName);
  if (uniqueArray(imageNames).length !== imageNames.length) errors.push('IMAGE_FILE_NAMEが重複しています。');
  const indexes = items.map(item => item.imageIndex).sort((a, b) => a - b);
  indexes.forEach((value, idx) => {
    if (value !== idx + 1) errors.push('IMAGE_INDEXは1から件数までの連番にしてください。');
  });
































  items.sort((a, b) => a.imageIndex - b.imageIndex);
  return { items, total: items.length, batchId: batchIds.length === 1 ? batchIds[0] : '', errors: uniqueArray(errors) };
}

  const seasonJp = read('SEASON_JP');
  const imagePromptVersionRaw = read('IMAGE_PROMPT_VERSION');
  const imagePromptVersion = imagePromptVersionRaw ? Number(imagePromptVersionRaw) : 3;
  const titleMatchMode = read('TITLE_MATCH_MODE');
  const titleLayoutLinesRaw = read('TITLE_LAYOUT_LINES');
  const textAllowlist = read('TEXT_ALLOWLIST');
  const extraTextForbidden = read('EXTRA_TEXT_FORBIDDEN');
  const contextScope = read('CONTEXT_SCOPE');
  const allowedElements = read('ALLOWED_ELEMENTS');
  const forbiddenElements = read('FORBIDDEN_ELEMENTS');
  const officialResemblanceGuard = read('OFFICIAL_RESEMBLANCE_GUARD');
  const styleProfile = read('STYLE_PROFILE');
































  const missing = [];
  if (!batchId) missing.push('BATCH_ID');
  if (!Number.isInteger(imageIndex) || imageIndex < 1 || imageIndex > ZIP_BATCH_MAX_ARTICLES) missing.push('IMAGE_INDEX');
  if (!workId) missing.push('WORK_ID');
  if (!packageFolderName) missing.push('PACKAGE_FOLDER_NAME');
  if (!imageFileName) missing.push('IMAGE_FILE_NAME');
  if (!title) missing.push('TITLE');
  if (!titleRenderRequired) missing.push('TITLE_RENDER_REQUIRED');
  if (!seasonJp) missing.push('SEASON_JP');
  if (missing.length) throw new Error('IMAGE_PROMPT管理行不足：' + missing.join(', '));
































  if (String(titleRenderRequired).toUpperCase() !== 'TRUE') {
    throw new Error('TITLE_RENDER_REQUIREDはTRUEにしてください。ブログタイトル全文の画像内描画が必須です');
  }


  // IMAGE_PROMPT_VERSION 4以降だけ新ガードを厳格検査する。既存V3 RUNは互換継続する。
  if (Number.isInteger(imagePromptVersion) && imagePromptVersion >= 4) {
    const v4Missing = [];
    if (String(titleMatchMode).toUpperCase() !== 'EXACT') v4Missing.push('TITLE_MATCH_MODE=EXACT');
    const titleLayoutLines = Number(titleLayoutLinesRaw);
    if (!Number.isInteger(titleLayoutLines) || titleLayoutLines < 1 || titleLayoutLines > 4) v4Missing.push('TITLE_LAYOUT_LINES=1..4');
    if (String(textAllowlist).toUpperCase() !== 'TITLE|MAIN_HOOK|SUPPORT_CALLOUTS|TARGET_NAME') v4Missing.push('TEXT_ALLOWLIST');
    if (String(extraTextForbidden).toUpperCase() !== 'TRUE') v4Missing.push('EXTRA_TEXT_FORBIDDEN=TRUE');
    if (String(contextScope).toUpperCase() !== 'CURRENT_IMAGE_INDEX_ONLY') v4Missing.push('CONTEXT_SCOPE=CURRENT_IMAGE_INDEX_ONLY');
    if (!allowedElements) v4Missing.push('ALLOWED_ELEMENTS');
    if (!forbiddenElements) v4Missing.push('FORBIDDEN_ELEMENTS');
    if (String(officialResemblanceGuard).toUpperCase() !== 'STRONG') v4Missing.push('OFFICIAL_RESEMBLANCE_GUARD=STRONG');
    if (String(styleProfile).toUpperCase() !== 'HNK_HIGH_IMPACT_EDITORIAL_V1') v4Missing.push('STYLE_PROFILE=HNK_HIGH_IMPACT_EDITORIAL_V1');
    if (v4Missing.length) throw new Error('IMAGE_PROMPT V4ガード不足：' + v4Missing.join(', '));


    if (Number.isInteger(titleLayoutLines) && titleLayoutLines >= 1 && titleLayoutLines <= 4) {
      const titleLines = [];
      for (let i = 1; i <= titleLayoutLines; i++) {
        const line = read('TITLE_LINE_' + i);
        if (!line) throw new Error('TITLE_LINE_' + i + ' がありません');
        titleLines.push(line);
      }
      const normalizeLayoutText = value => String(value || '').replace(/\s+/g, '');
      if (normalizeLayoutText(titleLines.join('')) !== normalizeLayoutText(title)) {
        throw new Error('TITLE_LINE_1〜' + titleLayoutLines + ' を連結した文字列がTITLEと一致しません');
      }
    }
  }


  const expectedImageFileName = workId + '_eyecatch.png';
  if (imageFileName !== expectedImageFileName) {
    throw new Error('IMAGE_FILE_NAMEは「' + expectedImageFileName + '」にしてください');
  }
































  return {
    batchId,
    imageIndex,
    workId,
    packageFolderName,
    imageFileName,
    title,
    titleRenderRequired,
    seasonJp,
    imagePromptVersion,
    titleMatchMode,
    titleLayoutLinesRaw,
    textAllowlist,
    extraTextForbidden,
    contextScope,
    allowedElements,
    forbiddenElements,
    officialResemblanceGuard,
    styleProfile,
    imagePrompt
  };
}

function stripCommonZipRoot_(paths) {

  while (current.length) {
    const split = current.map(path => path.split('/').filter(Boolean));
    if (split.some(parts => parts.length < 2)) break;
































    const first = split[0][0];
    if (!split.every(parts => parts[0] === first)) break;
































    current = split.map(parts => parts.slice(1).join('/'));
  }
































  return current;

/**
 * 明示されたバッチフォルダIDから保存済みbatch-state.jsonを直接復元する。
 * 事故復旧時に全履歴走査を避けるための高速経路。
 */
function loadBatchStateFromFolderHint_(batchId, batchFolderIdHint) {
  const folderId = String(batchFolderIdHint || '').trim();
  if (!folderId) return null;




  const batchFolder = DriveApp.getFolderById(folderId);
  if (String(batchFolder.getName() || '') !== batchId) {
    throw new Error('指定バッチフォルダとBATCH_IDが一致しません：' + batchId);
  }
  try { batchFolder.setTrashed(false); } catch (err) {
    console.error('指定バッチフォルダ復元失敗：' + err.message);
  }




  const stateFiles = batchFolder.getFilesByName('batch-state.json');
  if (!stateFiles.hasNext()) {
    throw new Error('指定バッチフォルダにbatch-state.jsonがありません：' + batchId);
  }
  const state = loadZipBatchState_(stateFiles.next().getId());
  if (String(state.batchId || '') !== batchId) {
    throw new Error('batch-state.jsonのBATCH_IDが一致しません：' + batchId);
  }
  return state;
}








/**
 * 記事台帳のDRIVE_ARTICLE_URLから、保存済みバッチのbatch-state.jsonを逆引きする。
 * finalize後にPENDING_BATCH_PREFIXが消えていても、同一BATCH_IDの再開を可能にする。
 */
function loadBatchStateFromLedger_(batchId) {
  try {
    const ledgerData = readArticleLedgerRows_();
    const h = {};
    ARTICLE_LEDGER_HEADERS.forEach((name, index) => { h[name] = index; });




    for (let i = 0; i < ledgerData.rows.length; i++) {
      const row = ledgerData.rows[i];
      if (String(row[h.BATCH_ID] || '') !== batchId) continue;




      const driveArticleUrl = String(row[h.DRIVE_ARTICLE_URL] || '');
      const idMatch = driveArticleUrl.match(/\/d\/([A-Za-z0-9_-]+)/) ||
        driveArticleUrl.match(/[?&]id=([A-Za-z0-9_-]+)/);
      if (!idMatch) continue;




      try {
        const txtFile = DriveApp.getFileById(idMatch[1]);
        const itemParents = txtFile.getParents();
        if (!itemParents.hasNext()) continue;
        const itemFolder = itemParents.next();




        const batchParents = itemFolder.getParents();
        if (!batchParents.hasNext()) continue;
        const batchFolder = batchParents.next();
        if (String(batchFolder.getName() || '') !== batchId) continue;




        try { batchFolder.setTrashed(false); } catch (restoreErr) {
          console.error('台帳逆引きバッチフォルダ復元失敗：' + restoreErr.message);
        }




        const stateFiles = batchFolder.getFilesByName('batch-state.json');
        if (!stateFiles.hasNext()) continue;
        const stateFile = stateFiles.next();
        const state = loadZipBatchState_(stateFile.getId());
        if (String(state.batchId || '') === batchId) return state;
      } catch (itemErr) {
        console.error('台帳逆引き候補の読込失敗：' + itemErr.message);
      }
    }
  } catch (err) {
    console.error('台帳からのバッチ状態逆引き失敗：' + err.message);
  }
  return null;
}








/**
 * ZIPバッチを1記事だけ進める。
 */
function resumeZipBatchById_(batchId, notifyChannel, notifyThreadTs, batchFolderIdHint) {
  batchId = String(batchId || '').trim();
  if (!batchId) throw new Error('BATCH_IDを指定してください');








  const props = PropertiesService.getScriptProperties();
  let state = loadBatchStateFromFolderHint_(batchId, batchFolderIdHint);
  if (!state) state = loadBatchStateFromLedger_(batchId);




  if (!state) {
    const all = props.getProperties();
    const candidates = [];
    const pendingStateFileId = props.getProperty(PENDING_BATCH_PREFIX + batchId);
    if (pendingStateFileId) candidates.push(pendingStateFileId);




    Object.keys(all).forEach(key => {
      if (key.indexOf(ZIP_BATCH_HASH_PREFIX) !== 0) return;
      const info = parseJsonSafe_(all[key], {});
      if (info.stateFileId && candidates.indexOf(info.stateFileId) === -1) {
        candidates.push(info.stateFileId);
      }
    });




    for (let i = 0; i < candidates.length; i++) {
      try {
        const loaded = loadZipBatchState_(candidates[i]);
        if (String(loaded.batchId || '') === batchId) {
          state = loaded;
          break;
        }
      } catch (err) {
        console.error('バッチ再開候補の読込失敗：' + err.message);
      }
    }
  }








  if (!state) {
    throw new Error('再開可能な保存済みバッチが見つかりません：' + batchId);
  }








  let resetCount = 0;
  (state.items || []).forEach(item => {
    const errorText = String(item.error || item.validationError || '');
    const lowerErrorText = errorText.toLowerCase();
    const isArticleLedgerHold = errorText.indexOf('ARTICLE_LEDGER_HOLD') !== -1;
    const isFotolifeCapacityError =
      errorText.indexOf('はてなフォトライフ画像アップロード失敗') !== -1 &&
      (errorText.indexOf('HTTP 402') !== -1 ||
       lowerErrorText.indexOf('fotolife disk is full') !== -1 ||
       lowerErrorText.indexOf('disk full') !== -1);




    if (item.status === 'failed' && (isArticleLedgerHold || isFotolifeCapacityError)) {
      item.status = 'pending';
      item.stage = isFotolifeCapacityError
        ? 'fotolife_capacity_requeued'
        : 'article_ledger_hold_requeued';
      item.attempts = 0;
      item.error = '';
      item.validationError = '';
      resetCount++;
    }
  });








  if (!resetCount && !(state.items || []).some(item => item.status === 'pending')) {
    throw new Error('再開可能な失敗記事がありません：' + batchId);
  }








  try {
    DriveApp.getFolderById(state.batchFolderId).setTrashed(false);
  } catch (err) {
    console.error('バッチフォルダ復元失敗：' + err.message);
  }








  state.status = 'queued';
  state.summarySent = false;
  state.nextAttemptAt = 0;
  state.articleLedgerHoldNotified = false;
  state.articleLedgerHoldError = '';
  // v41: 再開コマンドのスレッドへ最終結果を複製しない。完了通知は元TEXT ZIPスレッドを正本とする。
  state.resumeNotifyChannel = '';
  state.resumeNotifyThreadTs = '';
  state.resumeRequestedAt = new Date().toISOString();
  state.updatedAt = new Date().toISOString();
  saveZipBatchState_(state);








  setScriptPropertySafe_(PENDING_BATCH_PREFIX + batchId, state.stateFileId);
  props.deleteProperty(IMAGE_BATCH_ACCEPT_PREFIX + batchId);




  // 手動再開は既存キューの末尾追加ではなく先頭へ移し、同一tickで即再処理できるようにする。
  const resumeQueue = getZipBatchQueue_().filter(id => id !== state.stateFileId);
  resumeQueue.unshift(state.stateFileId);
  saveZipBatchQueue_(resumeQueue);








  const pendingCount = (state.items || []).filter(item => item.status === 'pending').length;
  const publishedCount = (state.items || []).filter(item => item.status === 'published').length;
  const failedCount = (state.items || []).filter(item => item.status === 'failed').length;
  return {
    batchId: batchId,
    resetCount: resetCount,
    pendingCount: pendingCount,
    publishedCount: publishedCount,
    failedCount: failedCount,
    status: state.status,
    stateFileId: state.stateFileId
  };
}








function rotateZipBatchStateToTail_(stateFileId) {
  const queue = getZipBatchQueue_().filter(id => id !== stateFileId);
  queue.push(stateFileId);
  saveZipBatchQueue_(queue);
}








function processZipBatchQueueBurst_() {
  const startedAt = Date.now();
  let handled = 0;




  while (handled < ZIP_BATCH_ITEMS_PER_TICK) {
    if (Date.now() - startedAt >= ZIP_BATCH_TICK_BUDGET_MS) break;
    const outcome = processNextZipBatchItem_();
    if (outcome === 'idle') break;
    handled++;
  }




  return handled;
}








function processNextZipBatchItem_() {
  const stateFileId = peekZipBatchState_();
  if (!stateFileId) return 'idle';








  let state;
  try {
    state = loadZipBatchState_(stateFileId);
  } catch (err) {
    console.error(err);
    removeZipBatchStateFromQueue_(stateFileId);
    return 'removed';
  }








  const nextAttemptAt = Number(state.nextAttemptAt || 0);
  if (nextAttemptAt > Date.now()) {
    rotateZipBatchStateToTail_(stateFileId);
    return 'blocked';
  }








  state.status = 'processing';
  state.updatedAt = new Date().toISOString();








  const item = state.items.find(entry => entry.status === 'pending');








  if (!item) {
    finalizeZipBatch_(state);
    return 'finalized';
  }








  try {
    processZipBatchArticle_(state, item);
    state.nextAttemptAt = 0;
    state.articleLedgerHoldNotified = false;
    state.articleLedgerHoldError = '';
    state.updatedAt = new Date().toISOString();
    saveZipBatchState_(state);
  } catch (err) {
    console.error(err);
    const errorMessage = String(err && err.message ? err.message : err);








    if (errorMessage.indexOf('ARTICLE_LEDGER_HOLD') !== -1) {
      item.error = errorMessage;
      item.stage = 'article_ledger_hold';
      item.status = 'pending';
      state.status = 'article_ledger_hold';
      state.articleLedgerHoldError = errorMessage;
      state.nextAttemptAt = Date.now() + (5 * 60 * 1000);
      state.updatedAt = new Date().toISOString();








      if (!state.articleLedgerHoldNotified) {
        postSlackMessage(
          state.channel,
          '記事台帳・既存記事確認エラーで公開を保留しました。文章・画像・バッチは削除せず保持し、自動再試行します。\nBATCH_ID：' + state.batchId + '\nERROR：' + trimForSlackError(errorMessage, 500),
          state.threadTs
        );
        state.articleLedgerHoldNotified = true;
      }








      saveZipBatchState_(state);
      rotateZipBatchStateToTail_(state.stateFileId);
      return 'blocked';
    }








    item.attempts = Number(item.attempts || 0) + 1;
    item.error = errorMessage;
    item.stage = 'retry_wait';








    if (item.attempts >= ZIP_BATCH_RETRY_LIMIT || isZipBatchPermanentError_(item.error)) {
      item.status = 'failed';
      item.stage = 'failed';
      state.nextAttemptAt = 0;
    } else {
      state.nextAttemptAt = Date.now() + ZIP_BATCH_RETRY_WAIT_MS;
    }








    state.updatedAt = new Date().toISOString();
    saveZipBatchState_(state);
  }








  if (!state.items.some(entry => entry.status === 'pending')) {
    finalizeZipBatch_(state);
    return 'finalized';
  }




  if (Number(state.nextAttemptAt || 0) > Date.now()) {
    rotateZipBatchStateToTail_(state.stateFileId);
    return 'blocked';
  }




  return 'processed';
}








function processZipBatchArticle_(state, item) {
  if (!item.imageFileId) throw new Error('対応画像が未登録です：' + item.expectedImageFileName);
  const txtFile = DriveApp.getFileById(item.txtFileId);
  const imageFile = DriveApp.getFileById(item.imageFileId);
  const articleText = txtFile.getBlob().getDataAsString('UTF-8').replace(/^\uFEFF/, '');
  const cleaned = normalizeArticleInput(articleText);
































  if (containsWrongBlogInfo(cleaned)) {
    throw new Error('別ブログ情報が混在しています');
  }
  if (!isArticleText(cleaned)) {
    throw new Error('必要な固定タグが不足しています');
  }
































  const parsed = parseArticleOutput(cleaned);
  if (!parsed.ok) {
    throw new Error(parsed.errors.join(' / '));
  }
































  item.title = parsed.title;
  item.articleHash = item.articleHash || computeTextSha256_(cleaned);




  const actionContext = resolveBatchArticleActionContext_(state, item);
  item.articleAction = actionContext.articleAction || item.articleAction || '';
  item.parentArticleKey = actionContext.parentArticleKey || item.parentArticleKey || '';
  item.relationNote = actionContext.relationNote || item.relationNote || '';




  const isUpdateExisting = item.articleAction === 'UPDATE' || item.articleAction === 'UPDATE_EXISTING';
  if (isUpdateExisting && !item.editUrl) {
    const updateTarget = resolveExistingUpdateTarget_(item.parentArticleKey);
    item.editUrl = updateTarget.editUrl;
    item.updateTargetPublicUrl = updateTarget.publicUrl;
    item.updateTargetWorkId = updateTarget.workId;
    item.stage = 'update_target_resolved';
    state.updatedAt = new Date().toISOString();
    saveZipBatchState_(state);
    safeSyncBatchStateToArticleLedger_(state);
  }




  const ledgerDuplicate = isUpdateExisting
    ? findPublishedLedgerWorkDuplicate_(item.workId)
    : findPublishedLedgerDuplicate_(parsed.title, item.workId, item.articleHash);
  if (ledgerDuplicate) {
    item.status = 'duplicate';
    item.stage = 'duplicate_skipped_by_ledger';
    item.publicUrl = ledgerDuplicate.publicUrl || '';
    item.error = '記事台帳の公開済み記録に一致したため重複公開を防止しました';
    state.updatedAt = new Date().toISOString();
    saveZipBatchState_(state);
    safeSyncBatchStateToArticleLedger_(state);
    return;
  }
































  const publishedRaw = getOptionalProp(ZIP_ARTICLE_HASH_PREFIX + item.articleHash);
  if (publishedRaw) {
    let publishedInfo = {};
    try { publishedInfo = JSON.parse(publishedRaw); } catch (err) {}
    item.status = 'duplicate';
    item.stage = 'duplicate_skipped';
    item.publicUrl = publishedInfo.publicUrl || '';
    item.error = '同一記事の重複公開を防止しました';
    state.updatedAt = new Date().toISOString();
    saveZipBatchState_(state);
    safeSyncBatchStateToArticleLedger_(state);
    return;
  }
































  if (!item.imageUrl) {
    const imageInfo = uploadToHatenaFotolife(imageFile.getBlob(), parsed.title);
    item.imageUrl = imageInfo.imageUrl;
    item.stage = 'image_uploaded';
    state.updatedAt = new Date().toISOString();
    saveZipBatchState_(state);
    safeSyncBatchStateToArticleLedger_(state);
  }
































  const imageHtml =
    '<div style="margin:20px 0; text-align:center;">' +
    '<img src="' + escapeHtml(item.imageUrl) + '" alt="' + escapeHtml(parsed.title) + '" style="max-width:100%; height:auto; border-radius:16px;">' +
    '</div>\n';
































  const finalHtml = imageHtml + parsed.html;
































  if (!item.editUrl) {
    const draftResult = createHatenaDraftStrictCategories_({
      title: parsed.title,
      html: finalHtml,
      categories: parsed.categories
    });
































    item.editUrl = draftResult.editUrl;
    item.previewUrl = draftResult.previewUrl || '';
    item.stage = 'draft_created';
    state.updatedAt = new Date().toISOString();
    saveZipBatchState_(state);
    safeSyncBatchStateToArticleLedger_(state);
  }
































  const result = updateHatenaEntry({
    editUrl: item.editUrl,
    title: parsed.title,
    html: finalHtml,
    categories: parsed.categories,
    draft: false,
    preview: false
  });
































  item.publicUrl = result.publicUrl || item.updateTargetPublicUrl || item.publicUrl || item.previewUrl || item.editUrl;
  item.status = 'published';
  item.stage = 'published';
  item.error = '';
  item.validationError = '';
  state.updatedAt = new Date().toISOString();
  saveZipBatchState_(state);
  safeSyncBatchStateToArticleLedger_(state);
































  setScriptPropertySafe_(
    ZIP_ARTICLE_HASH_PREFIX + item.articleHash,
    JSON.stringify({ publicUrl: item.publicUrl, title: item.title, ts: Date.now() })
  );
}
































function createHatenaDraftStrictCategories_({ title, html, categories }) {
  const hatenaId = getProp(CONFIG_KEYS.HATENA_ID);
  const blogId = getProp(CONFIG_KEYS.HATENA_BLOG_ID);
  const url =
    'https://blog.hatena.ne.jp/' +
    encodeURIComponent(hatenaId) + '/' +
    encodeURIComponent(blogId) +
    '/atom/entry';
































  const safeCategories = sanitizeHatenaCategories(categories);
  if (safeCategories.length !== categories.length || safeCategories.length < 1) {
    throw new Error('正式カテゴリを保持できないため公開を停止しました');
  }
































  const variants = [
    { name: 'full-original', html, categories: safeCategories },
    { name: 'no-iframe', html: replaceIframeWithSafeLink(html), categories: safeCategories },
    { name: 'minimal-no-iframe', html: makeHatenaMinimalHtml(replaceIframeWithSafeLink(html)), categories: safeCategories }
  ];
  const errors = [];
































  for (let i = 0; i < variants.length; i++) {
    const variant = variants[i];
    try {
      return createHatenaDraftOnce({
        url,
        title,
        html: variant.html,
        categories: variant.categories,
        attemptName: 'zip-' + variant.name
      });
    } catch (err) {
      const message = String(err && err.message ? err.message : err);
      errors.push('[' + variant.name + '] ' + trimForSlackError(message, 700));
      if (message.includes('HTTP 401') || message.includes('HTTP 403') || message.includes('HTTP 404')) {
        throw err;
      }
    }
  }
































  throw new Error('はてな下書き作成失敗：' + errors.join(' / '));
}
































function isZipBatchPermanentError_(message) {
  const text = String(message || '');
  return (

function finalizeZipBatch_(state) {
  if (state.summarySent) {
    removeZipBatchStateFromQueue_(state.stateFileId);
    return;
  }
































  const published = state.items.filter(item => item.status === 'published');
  const duplicates = state.items.filter(item => item.status === 'duplicate');
  const failed = state.items.filter(item => item.status === 'failed');
































  const lines = [
    'ZIP一括公開完了',
    '公開：' + published.length + '件',
    '重複防止：' + duplicates.length + '件',
    '失敗：' + failed.length + '件'
  ];
































  published.forEach(item => {
    lines.push('✅ ' + pad2_(item.index) + ' ' + (item.title || item.displayName) + '\n' + item.publicUrl);
  });
  duplicates.forEach(item => {
    lines.push('⏭️ ' + pad2_(item.index) + ' ' + (item.title || item.displayName) + '\n' + (item.publicUrl || '同一記事のため公開なし'));
  });
  failed.forEach(item => {
    lines.push('❌ ' + pad2_(item.index) + ' ' + (item.title || item.displayName) + '\n' + trimForSlackError(item.error || item.validationError || '不明なエラー', 500));
  });
































  const summaryMessage = lines.join('\n\n');
  // v41: BATCH_IDごとの公開完了通知は元TEXT ZIP親スレッドへ1回だけ送る。
  // 再開・復旧・状態確認などの補助スレッドへ同一URL一覧を複製しない。
  postSlackMessage(state.channel, summaryMessage, state.threadTs);




  // 旧stateとの後方互換用に残存値だけ消す。
  state.resumeNotifyChannel = '';
  state.resumeNotifyThreadTs = '';
































  state.status = failed.length ? 'completed_with_errors' : 'completed';
  state.summarySent = true;
  state.updatedAt = new Date().toISOString();
  saveZipBatchState_(state);
  safeSyncBatchStateToArticleLedger_(state);
































  setScriptPropertySafe_(
    ZIP_BATCH_HASH_PREFIX + state.zipHash,
    JSON.stringify({
      stateFileId: state.stateFileId,
      status: state.status,
      published: published.length,
      duplicates: duplicates.length,
      failed: failed.length,
      ts: Date.now()
    })
  );
































  removeZipBatchStateFromQueue_(state.stateFileId);
  PropertiesService.getScriptProperties().deleteProperty(PENDING_BATCH_PREFIX + state.batchId);
































  try {
    DriveApp.getFolderById(state.batchFolderId).setTrashed(true);
  } catch (err) {
    console.error(err);
  }
































  cleanupOldZipBatchHashKeys_();
}

  }
































  return false;

/**
 * Slackのfiles:read側が403/404で使えない場合の文章ZIP復旧経路。
 * Drive上へ退避した同一ZIPをFile ID指定で直接受理し、通常の文章ZIP検証・キュー登録へ流す。
 */
function handleDriveTextZipRecovery_(channel, threadTs, driveFileId) {
  try {
    const fileId = String(driveFileId || '').trim();
    if (!/^[A-Za-z0-9_-]+$/.test(fileId)) {
      throw new Error('Drive File IDが不正です。');
    }




    const driveFile = withDriveRetry_('Drive文章ZIPファイル取得 ' + fileId, function() {
      return DriveApp.getFileById(fileId);
    });
    const name = String(withDriveRetry_('Drive文章ZIP名取得 ' + fileId, function() { return driveFile.getName(); }) || '');
    if (!/^HIMANEKO_TEXT_BATCH_.*\.zip$/i.test(name)) {
      throw new Error('文章ZIPではありません：' + name);
    }




    const zipBlob = withDriveRetry_('Drive文章ZIP Blob取得 ' + fileId, function() {
      return driveFile.getBlob().setName(name);
    });
    const bytes = zipBlob.getBytes();
    if (!bytes.length) throw new Error('Drive上のZIPが空です。');
    if (bytes.length > ZIP_BATCH_MAX_BYTES) {
      throw new Error('ZIPは40MB以下にしてください。');
    }




    const pseudoSlackFile = {
      id: 'DRIVE_' + fileId + '_' + Date.now(),
      name: name,
      title: name,
      size: bytes.length,
      mimetype: 'application/zip',
      filetype: 'zip'
    };




    postSlackMessage(
      channel,
      'Drive退避ZIPから復旧処理を開始します：' + name,
      threadTs
    );




    return handleSlackZipFileMessage(
      channel,
      pseudoSlackFile,
      threadTs,
      zipBlob
    );
  } catch (err) {
    postSlackMessage(
      channel,
      '❌ Drive文章ZIP復旧エラー：' + String(err && err.message ? err.message : err),
      threadTs
    );
    return false;
  }
}




















/**
 * Slackのfiles:read側が403/404で使えない場合の画像ZIP復旧経路。
 * Drive上へ退避した同一IMAGE ZIPをFile ID指定で直接受理し、通常の画像ZIP検証・公開キューへ流す。
 */
function handleDriveImageZipRecovery_(channel, threadTs, driveFileId) {
  try {
    const fileId = String(driveFileId || '').trim();
    if (!/^[A-Za-z0-9_-]+$/.test(fileId)) {
      throw new Error('Drive File IDが不正です。');
    }




    const driveFile = DriveApp.getFileById(fileId);
    const name = String(driveFile.getName() || '');
    const identity = name.match(/^HIMANEKO_IMAGE_BATCH_(.+)_(10|[1-9])\.zip$/i);
    if (!identity) {
      throw new Error('画像ZIPではありません：' + name);
    }




    const batchId = String(identity[1] || '');
    let canonicalChannel = channel;
    let canonicalThreadTs = threadTs;
    const pendingStateFileId = getOptionalProp(PENDING_BATCH_PREFIX + batchId);
    if (pendingStateFileId) {
      try {
        const recoveryState = loadZipBatchState_(pendingStateFileId);
        canonicalChannel = String(recoveryState.channel || canonicalChannel);
        canonicalThreadTs = String(recoveryState.threadTs || canonicalThreadTs);
      } catch (stateErr) {
        console.error('Drive画像ZIP復旧の元スレッド解決失敗：' + stateErr.message);
      }
    }




    const zipBlob = driveFile.getBlob().setName(name);
    const bytes = zipBlob.getBytes();
    if (!bytes.length) throw new Error('Drive上のZIPが空です。');
    if (bytes.length > ZIP_BATCH_MAX_BYTES) {
      throw new Error('ZIPは40MB以下にしてください。');
    }




    const pseudoSlackFile = {
      id: 'DRIVE_IMAGE_' + fileId + '_' + Date.now(),
      name: name,
      title: name,
      size: bytes.length,
      mimetype: 'application/zip',
      filetype: 'zip'
    };




    const isSeparateRecoveryThread =
      canonicalChannel !== String(channel || '') ||
      canonicalThreadTs !== String(threadTs || '');
    if (isSeparateRecoveryThread) {
      postSlackMessage(
        channel,
        '画像Drive復旧受付\nBATCH_ID：' + batchId + '\n以後の処理結果は元TEXT ZIPスレッドへ集約します。',
        threadTs
      );
    }




    postSlackMessage(
      canonicalChannel,
      'Drive退避IMAGE ZIPから復旧処理を開始します：' + name,
      canonicalThreadTs
    );




    return handleSlackZipFileMessage(
      canonicalChannel,
      pseudoSlackFile,
      canonicalThreadTs,
      zipBlob
    );
  } catch (err) {
    postSlackMessage(
      channel,
      '❌ Drive画像ZIP復旧エラー：' + String(err && err.message ? err.message : err),
      threadTs
    );
    return false;
  }
}

/**
 * SlackファイルをBlobで取得
 */
function getSlackFileBlob(file) {
  const token = getProp(CONFIG_KEYS.SLACK_BOT_TOKEN);
  const originalFile = file || {};
  const fileId = String(originalFile.id || '');
  let currentFile = originalFile;
  let lastCode = 0;
  let lastBody = '';
  let lastRefresh = 'not_attempted';




  // 外部アップロード直後のイベント内URLは古いことがある。
  // 初回を含む全試行でFile IDからfiles.infoを引き直し、最新URLだけを使う。
  const retryDelaysMs = [0, 2000, 4000, 8000, 16000, 30000];




  for (let attempt = 0; attempt < retryDelaysMs.length; attempt++) {
    if (attempt > 0) Utilities.sleep(retryDelaysMs[attempt]);




    if (!fileId) {
      lastBody = 'Slack File IDがありません。';
      break;
    }




    try {
      const info = slackApiGet_('files.info', { file: fileId });
      if (!info || !info.file) throw new Error('files.infoにfileがありません。');
      currentFile = info.file;
      lastRefresh = 'ok';
    } catch (refreshErr) {
      lastRefresh = 'failed:' + String(refreshErr && refreshErr.message ? refreshErr.message : refreshErr).slice(0, 300);
      console.error(
        'Slack files.info再取得失敗 attempt=' + (attempt + 1) +
        '/' + retryDelaysMs.length + ' file_id=' + fileId + ': ' + lastRefresh
      );
      // 初回からイベント内URLを正規URLとして使わない。files.info失敗は次回再試行する。
      lastBody = 'Slack files.info取得失敗：' + lastRefresh;
      continue;
    }




    const url = String(currentFile.url_private_download || currentFile.url_private || '');
    if (!url) {
      lastCode = 0;
      lastBody = 'Slack files.infoに取得URLがありません。';
      continue;
    }




    const res = UrlFetchApp.fetch(url, {
      method: 'get',
      headers: { Authorization: 'Bearer ' + token },
      muteHttpExceptions: true
    });




    const code = res.getResponseCode();
    if (code >= 200 && code < 300) {
      return res.getBlob().setName(originalFile.name || currentFile.name || 'slack-file');
    }




    lastCode = code;
    lastBody = String(res.getContentText() || '').slice(0, 500);
    const retryable = code === 404 || code === 429 || code >= 500;
    console.error(
      'Slackファイル取得再試行対象 attempt=' + (attempt + 1) +
      '/' + retryDelaysMs.length + ' HTTP=' + code + ' file_id=' + fileId
    );
    if (!retryable) break;
  }




  throw new Error(
    'Slackファイル取得失敗: HTTP ' + lastCode +
    '（再試行済み / file_id=' + fileId + ' / files.info=' + lastRefresh + '）' +
    (lastBody ? '\n' + lastBody : '')
  );
}








/**

function postSlackMessage(channel, message, threadTs) {

  const payload = {
    channel: channel,
    text: truncateSlackText(message, 12000),
    mrkdwn: true,
    unfurl_links: false,
    unfurl_media: false
  };
  if (threadTs) payload.thread_ts = threadTs;
































  const res = UrlFetchApp.fetch('https://slack.com/api/chat.postMessage', {
    method: 'post',
    contentType: 'application/json; charset=utf-8',
    payload: JSON.stringify(payload),
    headers: {
      Authorization: 'Bearer ' + token
    },
    muteHttpExceptions: true
  });
































  const code = res.getResponseCode();
  const body = res.getContentText();
































  if (code < 200 || code >= 300) {
    console.error('Slack送信失敗: ' + code + '\n' + body);
    return false;
  }
































  let data = {};

  }
































  if (!data.ok) {
    console.error('Slack送信エラー: ' + body);

  }
































  return true;
}

  const uploadRes = UrlFetchApp.fetch(urlData.upload_url, {

    contentType: 'application/octet-stream',
    payload: bytes,
    muteHttpExceptions: true
  });
  if (uploadRes.getResponseCode() < 200 || uploadRes.getResponseCode() >= 300) {
    throw new Error('Slackファイル本体アップロード失敗：HTTP ' + uploadRes.getResponseCode());

  });
































  const completeBody = completeRes.getContentText();

  if (completeRes.getResponseCode() < 200 || completeRes.getResponseCode() >= 300 || !completeData.ok) {
    throw new Error('Slackファイル共有完了処理失敗：' + completeBody);
  }

/**
 * 記事本文の共通正規化
 * 注意：
 * X_POSTの絵文字・記号を壊さないため、ここではHTML実体参照の一括変換はしない
 */
const FIXED_ARTICLE_TAG_NAMES = [
  'TITLE_START', 'TITLE_END',
  'CATEGORY_START', 'CATEGORY_END',
  'META_DESCRIPTION_START', 'META_DESCRIPTION_END',
  'HTML_START', 'HTML_END',
  'X_POST_START', 'X_POST_END',
  'IMAGE_PROMPT_START', 'IMAGE_PROMPT_END'
];
































/**
 * 固定タグの括弧表記揺れを正規化する。
 * 本文中の文字は変更せず、1行全体が固定タグの場合だけ【TAG】へ統一する。
 */
function normalizeFixedArticleTagLines(raw) {
  const tagPattern = FIXED_ARTICLE_TAG_NAMES.join('|');
  const re = new RegExp(
    '^[\\t ]*(?:【|「|『|（|\\(|［|\\[)?[\\t ]*(' + tagPattern + ')[\\t ]*(?:】|」|』|）|\\)|］|\\])?[\\t ]*$',
    'gm'
  );
































  return String(raw || '').replace(re, '【$1】');
}
































function normalizeArticleInput(raw) {
  let text = String(raw || '');
































  text = normalizeFixedArticleTagLines(text.replace(/^\uFEFF/, '').trim());
































  // SlackやChatGPT等で ``` ... ``` に囲まれている場合を除去
  text = normalizeFixedArticleTagLines(
    text
      .replace(/^```[a-zA-Z0-9_-]*\n?/, '')
      .replace(/\n?```$/, '')
      .trim()
  );
































  // 全体が `...` の1組で囲まれている場合を除去
  if (text.startsWith('`') && text.endsWith('`')) {
    const unwrapped = normalizeFixedArticleTagLines(text.slice(1, -1).trim());
    if (unwrapped.includes('【META_DESCRIPTION_START】') || unwrapped.includes('【TITLE_START】')) {
      text = unwrapped;
    }
  }
































  // 冒頭の補助文字を除去
  text = text
    .replace(/^記事化[:：]?\s*/i, '')
    .replace(/^記事全文[:：]?\s*/i, '')
    .trim();
































  return normalizeFixedArticleTagLines(text);
}
































/**
 * HTML部分だけSlack/HTMLエスケープを戻す
 *
 * .txtファイルのHTMLは基本的にそのまま使う。
 * &amp; を無条件に & へ戻すと、Amazonリンクなどが壊れるため、
 * Slack本文直貼りで <div> が &lt;div&gt; になっている場合だけ復元する。
 */
function decodeHtmlSectionOnly(html) {
  let s = String(html || '');
































  const looksEscapedHtml = /&lt;\s*\/?\s*(div|h1|h2|h3|p|ul|li|a|iframe|img|table|tr|th|td)\b/i.test(s);
































  if (!looksEscapedHtml) {
    return s;
  }
































  return s
    .replace(/&lt;/g, '<')
    .replace(/&gt;/g, '>')
    .replace(/&quot;/g, '"')
    .replace(/&#39;/g, "'")
    .replace(/&apos;/g, "'")
    .replace(/&amp;/g, '&');
}
































/**
 * XMLで使えない制御文字を除去する。
 * 絵文字、日本語、改行、タブは残す。
 */
function stripInvalidXmlChars(str) {
  return String(str || '').replace(/[\u0000-\u0008\u000B\u000C\u000E-\u001F]/g, '');
}
































/**
 * はてな本文HTMLの最終調整。
 * 生の & はHTMLとして危ないため &amp; に寄せる。
 * すでに &amp; などになっているものは壊さない。
 */
function sanitizeArticleHtmlForHatena(html) {
  let s = stripInvalidXmlChars(html)
    .replace(/\r\n/g, '\n')
    .replace(/\r/g, '\n')
    .trim();
































  s = s.replace(/&(?!amp;|lt;|gt;|quot;|apos;|nbsp;|#\d+;|#x[0-9A-Fa-f]+;|[A-Za-z][A-Za-z0-9]+;)/g, '&amp;');
































  return s;
}

/**
 * h1固定タグ出力を分解
 * 必須タグ：TITLE / CATEGORY / META_DESCRIPTION / HTML / X_POST / IMAGE_PROMPT
 * 旧タグ：META / X_POST_GAS / CHECK は使わない。
 */
function parseArticleOutput(raw) {
  const text = normalizeArticleInput(raw);
































  const title = extractBetween(text, '【TITLE_START】', '【TITLE_END】');
  const categoryRaw = extractBetween(text, '【CATEGORY_START】', '【CATEGORY_END】');
  const metaDescription = extractBetween(text, '【META_DESCRIPTION_START】', '【META_DESCRIPTION_END】');
  const htmlRaw = extractBetween(text, '【HTML_START】', '【HTML_END】');
  const xPostRaw = extractBetween(text, '【X_POST_START】', '【X_POST_END】');
  const imagePromptRaw = extractBetween(text, '【IMAGE_PROMPT_START】', '【IMAGE_PROMPT_END】');
































  const html = sanitizeArticleHtmlForHatena(decodeHtmlSectionOnly(htmlRaw));

  const categories = categoryResult.categories;
































  const errors = [];
































  if (!title) errors.push('TITLEがありません');
  if (!categoryRaw) errors.push('CATEGORYがありません');
  if (!metaDescription) errors.push('META_DESCRIPTIONがありません');
  if (!html) errors.push('HTMLがありません');
  if (!x_post) errors.push('X_POSTがありません');
  if (!image_prompt) errors.push('IMAGE_PROMPTがありません');
































  if (html && !html.includes('<div')) errors.push('HTML内に <div> がありません');
  if (html && !html.includes('</div>')) errors.push('HTML内に </div> がありません');
































  categoryResult.errors.forEach(message => errors.push(message));
































  if (!categories.length && !categoryResult.errors.length) {
    errors.push('はてなブログカテゴリーを取得できません');
  }
































  return {
    title,
    category_raw: categoryRaw,
    meta_description: metaDescription,
    html,
    x_post,
    image_prompt,
    categories,
    ok: errors.length === 0,
    errors
  };
}
































function parseHatenaCategoriesFromCategoryTag(categoryText) {
  if (!categoryText) return { categories: [], errors: [] };
































  const raw = String(categoryText || '')
    .replace(/\r\n/g, '\n')
    .replace(/\r/g, '\n');
































  const candidates = raw
    .split(/[\n,、]+/)
    .map(cat => cat.trim())
    .map(cat => cat.replace(/^[-・\s]+/, '').trim())
    .map(cat => cat.replace(/^カテゴリ\s*[：:]\s*/i, '').trim())
    .map(cat => cat.replace(/^Category\s*[：:]\s*/i, '').trim())
    .filter(Boolean);
































  if (!candidates.length) {
    return { categories: [], errors: ['CATEGORYが空です'] };
  }
































  const categories = [];
  const unknown = [];
































  candidates.forEach(item => {
    if (ALLOWED_HATENA_CATEGORIES.indexOf(item) !== -1) {
      categories.push(item);
      return;
    }
































    const mapped = ARTICLE_TYPE_CATEGORY_MAP[item];
    if (mapped && mapped.length) {
      mapped.forEach(cat => categories.push(cat));
      return;
    }
































    unknown.push(item);
  });
































  if (unknown.length) {
    return {
      categories: [],
      errors: ['正式カテゴリ一覧にないCATEGORYがあります：' + uniqueArray(unknown).join(' / ')]
    };
  }
































  const uniqueCategories = uniqueArray(categories);
































  if (uniqueCategories.length !== categories.length) {
    const duplicates = uniqueArray(
      categories.filter((cat, index) => categories.indexOf(cat) !== index)
    );
































    return {
      categories: [],
      errors: ['CATEGORYに重複があります：' + duplicates.join(' / ')]
    };
  }
































  if (uniqueCategories.length < 1 || uniqueCategories.length > 3) {
    return {
      categories: [],
      errors: ['CATEGORYは正式カテゴリを1〜3個だけ指定してください']
    };
  }
































  return { categories: uniqueCategories, errors: [] };
}

  return parseHatenaCategoriesFromCategoryTag(categoryText).categories;
}
































function extractBetween(str, start, end) {
  const s = str.indexOf(start);
  if (s === -1) return '';
  const from = s + start.length;
  const e = str.indexOf(end, from);
  if (e === -1) return '';
  return str.slice(from, e).trim();
}
































function extractHatenaCategories(meta) {
  if (!meta) return [];
































  const direct = extractMetaLineValue(meta, 'はてなブログカテゴリー') ||
                 extractMetaLineValue(meta, '推奨カテゴリー') ||
                 extractMetaLineValue(meta, 'Hatena Blog categories') ||
                 extractMetaLineValue(meta, 'Hatena categories') ||
                 extractMetaLineValue(meta, 'Categories');
































  if (direct) {
    const categories = direct
      .split(',')
      .map(cat => cat.trim())
      .filter(Boolean)
      .filter(cat => ALLOWED_HATENA_CATEGORIES.indexOf(cat) !== -1);
































    if (categories.length) {
      return uniqueArray(categories).slice(0, 3);
    }
  }
































  const genre = extractMetaLineValue(meta, '記事ジャンル');
  if (genre) {
    const matched = [];
































    Object.keys(ARTICLE_TYPE_CATEGORY_MAP).forEach(key => {
      if (genre.indexOf(key) !== -1) {
        ARTICLE_TYPE_CATEGORY_MAP[key].forEach(cat => matched.push(cat));
      }
    });
































    const safeMatched = uniqueArray(matched)
      .filter(cat => ALLOWED_HATENA_CATEGORIES.indexOf(cat) !== -1);
































    if (safeMatched.length) return safeMatched.slice(0, 3);
  }
































  const articleType = extractMetaLineValue(meta, '記事タイプ') ||
                      extractMetaLineValue(meta, '記事分類') ||
                      extractMetaLineValue(meta, 'Article type') ||
                      extractMetaLineValue(meta, 'Article classification');
































  if (articleType && ARTICLE_TYPE_CATEGORY_MAP[articleType]) {
    return ARTICLE_TYPE_CATEGORY_MAP[articleType];
  }
































  return ['エミュレータ最新情報'];
}

  const match = String(meta || '').match(pattern);
  return match ? match[1].trim() : '';
}

/**
 * はてなブログ下書き作成
 */
function createHatenaDraft({ title, html, categories }) {
  const hatenaId = getProp(CONFIG_KEYS.HATENA_ID);
  const blogId = getProp(CONFIG_KEYS.HATENA_BLOG_ID);
































  const url =
    'https://blog.hatena.ne.jp/' +
    encodeURIComponent(hatenaId) +
    '/' +
    encodeURIComponent(blogId) +
    '/atom/entry';
































  const variants = buildHatenaDraftVariants(html, categories);
  const errors = [];
































  for (let i = 0; i < variants.length; i++) {
    const variant = variants[i];
































    try {
      return createHatenaDraftOnce({
        url,
        title,
        html: variant.html,
        categories: variant.categories,
        attemptName: variant.name
      });
    } catch (err) {
      const message = String(err && err.message ? err.message : err);
      errors.push('[' + variant.name + '] ' + trimForSlackError(message, 900));
































      // 認証やURLの問題は、HTMLを軽量化しても直らないので即終了。
      if (message.includes('HTTP 401') || message.includes('HTTP 403') || message.includes('HTTP 404')) {
        throw err;
      }
    }
  }
































  throw new Error(
    'はてな下書き作成失敗: すべてのHTMLパターンで失敗しました。\n\n' +
    'この記事はタグ抽出までは成功しています。原因候補は、本文HTML内のiframe、inline style、カテゴリ、またははてな側のHTML制限です。\n\n' +
    errors.join('\n\n')
  );
}

function replaceIframeWithSafeLink(html) {
  let s = String(html || '');
































  function iframeToLink(match) {
    const srcMatch = String(match || '').match(/\ssrc=["']([^"']+)["']/i);
    const src = srcMatch ? normalizeYouTubeUrl(srcMatch[1]) : '';
































    if (!src) {
      return '<p>YouTube reference: check the official or trusted video linked in the source section.</p>';
    }
































    return '<p><a href="' + escapeHtmlAttribute(src) + '" target="_blank" rel="noopener noreferrer">YouTube reference video</a></p>';
  }
































  // iframeだけを含むレスポンシブdivは丸ごと置換。
  s = s.replace(/<div\b[^>]*>\s*<iframe\b[\s\S]*?<\/iframe>\s*<\/div>/gi, iframeToLink);
































  // 念のため単体iframeも置換。
  s = s.replace(/<iframe\b[\s\S]*?<\/iframe>/gi, iframeToLink);
































  return s;
}
































function normalizeYouTubeUrl(url) {
  let s = String(url || '').trim();
  s = s.replace(/&amp;/g, '&');
































  const embedMatch = s.match(/^https:\/\/www\.youtube\.com\/embed\/([^?&/]+)/i);
  if (embedMatch) {
    return 'https://www.youtube.com/watch?v=' + encodeURIComponent(embedMatch[1]);
  }
































  if (/^https:\/\/www\.youtube\.com\//i.test(s) || /^https:\/\/youtu\.be\//i.test(s)) {
    return s;
  }
































  return '';
}
































function makeHatenaMinimalHtml(html) {
  let s = String(html || '');
































  // はてな側のHTML制限に引っかかりやすい属性を削る。
  s = s
    .replace(/\sstyle=("[^"]*"|'[^']*')/gi, '')
    .replace(/\sallowfullscreen(=("[^"]*"|'[^']*'))?/gi, '')
    .replace(/\sallow=("[^"]*"|'[^']*')/gi, '')
    .replace(/\sframeborder=("[^"]*"|'[^']*')/gi, '')
    .replace(/\swidth=("[^"]*"|'[^']*')/gi, '')
    .replace(/\sheight=("[^"]*"|'[^']*')/gi, '');
































  // target/relは通常問題ないが、最小版では投稿成功を優先して削る。
  s = s
    .replace(/\starget=("[^"]*"|'[^']*')/gi, '')
    .replace(/\srel=("[^"]*"|'[^']*')/gi, '');
































  // Amazon検索URLの & はHTMLとして安全な &amp; に戻す。
  s = s.replace(/&(?!amp;|lt;|gt;|quot;|apos;|nbsp;|#\d+;|#x[0-9A-Fa-f]+;|[A-Za-z][A-Za-z0-9]+;)/g, '&amp;');
































  return s.trim();
}

    .replace(/>/g, '&gt;');

/**
 * はてなフォトライフ用に画像BlobのContent-Typeを補正
 * Slackから取得した画像が application/octet-stream になる問題を防ぐ
 */
function normalizeImageBlobForHatena(blob, fallbackTitle) {
  const bytes = blob.getBytes();
  const name = String(blob.getName() || fallbackTitle || 'eyecatch').toLowerCase();
































  let contentType = String(blob.getContentType() || '').toLowerCase();
































  // 拡張子から判定
  if (!contentType || contentType === 'application/octet-stream' || contentType === 'application/x-empty') {
    if (name.endsWith('.png')) {
      contentType = 'image/png';
    } else if (name.endsWith('.jpg') || name.endsWith('.jpeg')) {
      contentType = 'image/jpeg';
    } else if (name.endsWith('.webp')) {
      contentType = 'image/webp';
    }
  }
































  // バイナリ先頭から判定
  if (!contentType || contentType === 'application/octet-stream' || contentType === 'application/x-empty') {
    if (
      bytes.length >= 8 &&
      bytes[0] === 0x89 &&
      bytes[1] === 0x50 &&
      bytes[2] === 0x4E &&
      bytes[3] === 0x47
    ) {
      contentType = 'image/png';
    } else if (
      bytes.length >= 3 &&
      bytes[0] === 0xFF &&
      bytes[1] === 0xD8 &&
      bytes[2] === 0xFF
    ) {
      contentType = 'image/jpeg';
    } else if (
      bytes.length >= 12 &&
      bytes[0] === 0x52 &&
      bytes[1] === 0x49 &&
      bytes[2] === 0x46 &&
      bytes[3] === 0x46 &&
      bytes[8] === 0x57 &&
      bytes[9] === 0x45 &&
      bytes[10] === 0x42 &&
      bytes[11] === 0x50
    ) {
      contentType = 'image/webp';
    }
  }
































  // はてなフォトライフで安全に通すため、webpは一旦拒否
  if (contentType === 'image/webp') {
    throw new Error('webp画像ははてなフォトライフで失敗する可能性があります。png または jpg で送り直してください。');
  }
































  if (contentType !== 'image/png' && contentType !== 'image/jpeg') {
    throw new Error(
      '画像形式を判定できませんでした。\n' +
      'png または jpg 画像を送ってください。\n\n' +
      '現在のContent-Type: ' + contentType
    );
  }
































  let safeName = String(blob.getName() || 'eyecatch');
































  if (contentType === 'image/png' && !safeName.toLowerCase().endsWith('.png')) {
    safeName += '.png';
  }
































  if (contentType === 'image/jpeg' && !safeName.toLowerCase().match(/\.(jpg|jpeg)$/)) {
    safeName += '.jpg';
  }
































  return Utilities.newBlob(bytes, contentType, safeName);
}
































/**
 * はてなフォトライフへ画像アップロード
 */
function uploadToHatenaFotolife(blob, title) {
  const folder = getOptionalProp(CONFIG_KEYS.FOTOLIFE_FOLDER) || 'HimanekoBlog';
































  const fixedBlob = normalizeImageBlobForHatena(blob, title);
































  const contentType = fixedBlob.getContentType();
  const base64 = Utilities.base64Encode(fixedBlob.getBytes());
































  const xml =
    '<?xml version="1.0" encoding="utf-8"?>' +
    '<entry xmlns="http://purl.org/atom/ns#" xmlns:dc="http://purl.org/dc/elements/1.1/">' +
    '<title>' + escapeXml(title) + '</title>' +
    '<content mode="base64" type="' + escapeXml(contentType) + '">' + base64 + '</content>' +
    '<dc:subject>' + escapeXml(folder) + '</dc:subject>' +
        '<generator>GAS-Slack-Hatena-Himaneko-Himatsubushi</generator>' +
    '</entry>';
































  const res = UrlFetchApp.fetch('https://f.hatena.ne.jp/atom/post', {
    method: 'post',
    contentType: 'application/x.atom+xml; charset=utf-8',
    payload: xml,
    headers: {
      'X-WSSE': getHatenaWsseHeader()
    },
    muteHttpExceptions: true
  });
































  const code = res.getResponseCode();
  const body = res.getContentText();
































  if (code < 200 || code >= 300) {
    throw new Error(
      'はてなフォトライフ画像アップロード失敗: HTTP ' + code + '\n\n' +
      'Content-Type:\n' + contentType + '\n\n' +
      'レスポンス:\n' + body
    );
  }
































  const imageUrl = extractTag(body, 'hatena:imageurl');
  const syntax = extractTag(body, 'hatena:syntax');
































  if (!imageUrl) {
    throw new Error('はてなフォトライフの画像URL取得に失敗しました。\n' + body);
  }
































  return {
    imageUrl,
    syntax,
    responseXml: body
  };
}

/**
 * XML内link抽出
 */
function extractLink(xml, rel) {
  const pattern = new RegExp('<link[^>]+rel="' + rel + '"[^>]+href="([^"]+)"', 'i');
  const match = xml.match(pattern);
  return match ? decodeXml(match[1]) : '';
}
































/**
 * XMLタグ抽出
 */
function extractTag(xml, tagName) {
  const escaped = tagName.replace(':', '\\:');
  const pattern = new RegExp('<' + escaped + '[^>]*>([\\s\\S]*?)<\\/' + escaped + '>', 'i');
  const match = xml.match(pattern);
  return match ? decodeXml(match[1].trim()) : '';
}

function ledgerStatusRank_(status) {
  const ranks = {SELECTED:10,TEXT_CREATED:20,TEXT_VALIDATED:30,WAITING_IMAGES:40,IMAGE_CREATED:50,IMAGE_RECEIVED:60,READY_TO_PUBLISH:70,DRAFT:80,DUPLICATE:85,PUBLISHED:100};

}
function mergeLedgerStatus_(current, incoming) {
  current = String(current || '').toUpperCase(); incoming = String(incoming || '').toUpperCase();
  if (!incoming || current === 'PUBLISHED') return current;
  if (incoming === 'PUBLISHED') return incoming;
  if (['FAILED','PUBLISH_FAILED','IMAGE_HOLD','ARTICLE_HOLD','ABORTED'].indexOf(incoming) !== -1) return incoming;
  return ledgerStatusRank_(incoming) >= ledgerStatusRank_(current) ? incoming : current;
}
function buildArticleLedgerKey_(record) {
  if (record.WORK_ID) return 'WORK:' + record.WORK_ID;
  if (record.PUBLIC_URL) return 'URL:' + normalizeLedgerUrl_(record.PUBLIC_URL);
  if (record.EDIT_URL) return 'EDIT:' + normalizeLedgerUrl_(record.EDIT_URL);
  if (record.ARTICLE_HASH) return 'HASH:' + record.ARTICLE_HASH;
  return 'TITLE:' + normalizeLedgerTitleKey_(record.TITLE);
}

function findLedgerRowIndex_(rows, record, h) {
  const key = buildArticleLedgerKey_(record), title = normalizeLedgerTitleKey_(record.TITLE);
  for (let i = 0; i < rows.length; i++) {
    if (String(rows[i][h.ARTICLE_KEY] || '') === key) return i;
    if (record.WORK_ID && String(rows[i][h.WORK_ID] || '') === String(record.WORK_ID)) return i;
    if (record.PUBLIC_URL && normalizeLedgerUrl_(rows[i][h.PUBLIC_URL]) === normalizeLedgerUrl_(record.PUBLIC_URL)) return i;
    if (record.EDIT_URL && normalizeLedgerUrl_(rows[i][h.EDIT_URL]) === normalizeLedgerUrl_(record.EDIT_URL)) return i;
    if (record.ARTICLE_HASH && String(rows[i][h.ARTICLE_HASH] || '') === String(record.ARTICLE_HASH)) return i;
    if (title && normalizeLedgerTitleKey_(rows[i][h.TITLE]) === title) return i;
  }
  return -1;
}




function findLedgerUpsertRowIndex_(rows, record, h) {
  const key = buildArticleLedgerKey_(record);
  if (record.WORK_ID) {
    for (let i = 0; i < rows.length; i++) {
      if (String(rows[i][h.ARTICLE_KEY] || '') === key) return i;
      if (String(rows[i][h.WORK_ID] || '') === String(record.WORK_ID)) return i;
    }
    return -1;
  }
  return findLedgerRowIndex_(rows, record, h);
}




function upsertArticleLedgerRecords_(records, action) {
  const loaded = readArticleLedgerRows_(), rows = loaded.rows, h = ledgerHeaderIndex_();
  let inserted = 0, updated = 0;
  const touchedIndexes = {};
  (records || []).filter(Boolean).forEach(record => {
    let index = findLedgerUpsertRowIndex_(rows, record, h);
    if (index < 0) { index = rows.length; rows.push(ARTICLE_LEDGER_HEADERS.map(() => '')); inserted++; } else updated++;
    const row = rows[index], now = new Date().toISOString();
    const values = Object.assign({}, record, {
      ARTICLE_KEY:buildArticleLedgerKey_(record),
      NORMALIZED_TITLE:record.NORMALIZED_TITLE || normalizeLedgerTitleKey_(record.TITLE),
      UPDATED_AT:record.UPDATED_AT || now
    });
    Object.keys(values).forEach(header => {
      if (h[header] === undefined || values[header] === undefined || values[header] === null || String(values[header]) === '') return;
      row[h[header]] = header === 'STATUS' ? mergeLedgerStatus_(row[h.STATUS], values[header]) : values[header];
    });
    if (!row[h.CREATED_AT]) row[h.CREATED_AT] = record.CREATED_AT || now;
    touchedIndexes[index] = true;
  });
  const sheet = loaded.ctx.ledger;
  if (sheet.getMaxRows() < rows.length + 1) sheet.insertRowsAfter(sheet.getMaxRows(), rows.length + 1 - sheet.getMaxRows());
  Object.keys(touchedIndexes).map(Number).sort((a,b) => a-b).forEach(index => {
    sheet.getRange(index + 2, 1, 1, ARTICLE_LEDGER_HEADERS.length).setValues([rows[index]]);
  });
  loaded.ctx.logs.appendRow([new Date().toISOString(),action || 'UPSERT','SUCCESS',(records || []).length,'inserted='+inserted+', updated='+updated+', targeted_rows='+Object.keys(touchedIndexes).length]);
  return {total:rows.length, inserted:inserted, updated:updated};
}

}

function resolveHatenaEditUrlByPublicUrl_(publicUrl) {
  const target = normalizeLedgerUrl_(publicUrl);
  if (!target) return '';
  const hatenaId = getProp(CONFIG_KEYS.HATENA_ID);
  const blogId = getProp(CONFIG_KEYS.HATENA_BLOG_ID);
  let url = 'https://blog.hatena.ne.jp/' + encodeURIComponent(hatenaId) + '/' + encodeURIComponent(blogId) + '/atom/entry';
  const visited = {};
  let pages = 0;
  while (url) {
    if (visited[url] || ++pages > 500) throw new Error('UPDATE_EXISTING_TARGET：はてなAtomPubページング異常');
    visited[url] = true;
    const res = fetchHatenaLedgerPage_(url);
    const code = res.getResponseCode();
    const body = res.getContentText('UTF-8');
    if (code < 200 || code >= 300) throw new Error('UPDATE_EXISTING_TARGET：はてな記事一覧取得失敗 HTTP ' + code);
    const feed = XmlService.parse(stripHatenaAtomContentForMetadataParse_(body)).getRootElement();
    const ns = feed.getNamespace();
    const entries = feed.getChildren('entry', ns);
    for (let i = 0; i < entries.length; i++) {
      const alternate = atomLinkHref_(entries[i], 'alternate', ns);
      if (normalizeLedgerUrl_(alternate) === target) return atomLinkHref_(entries[i], 'edit', ns);
    }
    url = atomLinkHref_(feed, 'next', ns);
  }
  return '';
}




function resolveExistingUpdateTarget_(parentArticleKey) {
  const key = String(parentArticleKey || '').trim();
  if (!key) throw new Error('UPDATE_EXISTING_TARGET：PARENT_ARTICLE_KEYがありません');
  try {
    const loaded = readArticleLedgerRows_(), h = ledgerHeaderIndex_();
    const parentWorkId = key.indexOf('WORK:') === 0 ? key.slice(5) : '';
    const index = loaded.rows.findIndex(row =>
      String(row[h.ARTICLE_KEY] || '') === key ||
      (parentWorkId && String(row[h.WORK_ID] || '') === parentWorkId)
    );
    if (index < 0) throw new Error('UPDATE_EXISTING_TARGET：親記事が記事台帳にありません：' + key);
    const row = loaded.rows[index];
    const status = String(row[h.STATUS] || '').toUpperCase();
    if (status !== 'PUBLISHED') throw new Error('UPDATE_EXISTING_TARGET：親記事がPUBLISHEDではありません：' + status);
    const publicUrl = String(row[h.PUBLIC_URL] || '').trim();
    if (!publicUrl) throw new Error('UPDATE_EXISTING_TARGET：親記事のPUBLIC_URLがありません：' + key);
    let editUrl = String(row[h.EDIT_URL] || '').trim();
    if (!editUrl) {
      editUrl = resolveHatenaEditUrlByPublicUrl_(publicUrl);
      if (!editUrl) throw new Error('UPDATE_EXISTING_TARGET：親記事のEDIT_URLをHatena AtomPubから取得できません：' + publicUrl);
      if (parentWorkId) {
        upsertArticleLedgerRecords_([{WORK_ID:parentWorkId, EDIT_URL:editUrl, UPDATED_AT:new Date().toISOString()}], 'RESOLVE_UPDATE_EDIT_URL');
      }
    }
    return {articleKey:key, workId:parentWorkId || String(row[h.WORK_ID] || ''), title:String(row[h.TITLE] || ''), publicUrl:publicUrl, editUrl:editUrl};
  } catch (err) {
    const message = String(err && err.message ? err.message : err);
    if (message.indexOf('UPDATE_EXISTING_TARGET') !== -1) throw err;
    throw new Error('ARTICLE_LEDGER_HOLD：既存更新対象の解決に失敗しました。' + message);
  }
}




function atomLinkHref_(element, rel, ns) {
  const links = element.getChildren('link', ns);
  for (let i=0;i<links.length;i++) {
    const r=links[i].getAttribute('rel'), href=links[i].getAttribute('href');
    if (r && r.getValue() === rel) return href ? href.getValue() : '';
  }
  return '';
}
function fetchHatenaLedgerPage_(url) {
  const normalizedUrl = String(url || '').replace(/&amp;/g, '&');
  let lastError = '';




  for (let attempt = 1; attempt <= 2; attempt++) {
    try {
      return UrlFetchApp.fetch(normalizedUrl, {
        method: 'get',
        headers: { Authorization: getHatenaBasicAuthHeader() },
        muteHttpExceptions: true
      });
    } catch (err) {
      lastError = String(err && err.message ? err.message : err);
      if (attempt < 2) Utilities.sleep(1000);
    }
  }




  throw new Error('はてな記事一覧取得失敗: ' + lastError + '\n' + normalizedUrl);
}

    const feed=XmlService.parse(stripHatenaAtomContentForMetadataParse_(body)).getRootElement(), ns=feed.getNamespace(), appNs=XmlService.getNamespace('app','http://www.w3.org/2007/app');
    feed.getChildren('entry',ns).forEach(entry => {
      const title=entry.getChild('title',ns), published=entry.getChild('published',ns), updated=entry.getChild('updated',ns);
      const control=entry.getChild('control',appNs), draftEl=control ? control.getChild('draft',appNs) : null;
      const draft=!!(draftEl && String(draftEl.getText()).toLowerCase()==='yes');
      const categories=entry.getChildren('category',ns).map(c => c.getAttribute('term')).filter(Boolean).map(a => a.getValue());
      const publishedAt=published ? published.getText() : '', updatedAt=updated ? updated.getText() : '';
      records.push({STATUS:draft?'DRAFT':'PUBLISHED',TITLE:title?title.getText():'',CATEGORIES:categories.join(' / '),

    });
    url=atomLinkHref_(feed,'next',ns);
  }
  return {records:records,pages:pages};
}

function h3RepairFindLedgerWork_(workId) {
  const ss = SpreadsheetApp.openById(ARTICLE_LEDGER_SPREADSHEET_ID);
  const sh = ss.getSheetByName(ARTICLE_LEDGER_SHEET_NAME);
  const values = sh.getDataRange().getValues();
  if (!values.length) return null;
  const headers = values[0].map(function(v) { return String(v || ''); });
  const workCol = headers.indexOf('WORK_ID');
  if (workCol < 0) return null;




  for (let r = 1; r < values.length; r++) {
    if (String(values[r][workCol] || '') !== String(workId || '')) continue;
    const out = {};
    headers.forEach(function(header, c) { out[header] = values[r][c]; });
    out.__row = r + 1;
    return out;
  }
  return null;
}








function h3RepairUpdateLedgerWork_(workId, fields) {
  const ss = SpreadsheetApp.openById(ARTICLE_LEDGER_SPREADSHEET_ID);
  const sh = ss.getSheetByName(ARTICLE_LEDGER_SHEET_NAME);
  const values = sh.getDataRange().getValues();
  if (!values.length) throw new Error('記事台帳が空です');




  const headers = values[0].map(function(v) { return String(v || ''); });
  const workCol = headers.indexOf('WORK_ID');
  if (workCol < 0) throw new Error('記事台帳にWORK_ID列がありません');




  let row = -1;
  for (let r = 1; r < values.length; r++) {
    if (String(values[r][workCol] || '') === String(workId || '')) {
      row = r + 1;
      break;
    }
  }
  if (row < 0) throw new Error('記事台帳のWORK_IDが見つかりません: ' + workId);




  Object.keys(fields || {}).forEach(function(header) {
    const col = headers.indexOf(header);
    if (col < 0) return;
    sh.getRange(row, col + 1).setValue(fields[header]);
  });
}
